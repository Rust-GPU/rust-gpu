18/// Parses field invariants in attribute order.
19pub(crate) fn parse(attrs: &[Attribute]) -> Result<Vec<Expr>, Error> {
20    let mut invariants = Vec::new();
21    for attr in attrs.iter().filter(|attr| path_is_ident(attr.path(), "zerocopy")) {
22        let options = attr.parse_args_with(Punctuated::<Meta, Token![,]>::parse_terminated)?;
23        for option in options {
24            if !path_is_ident(option.path(), "invariant") {
25                return Err(Error::new_spanned(option, "expected `invariant(...)`"));
26            }
27            match option {
28                Meta::List(list) if matches!(list.delimiter, MacroDelimiter::Paren(_)) => {
29                    invariants.push(list.parse_args::<Expr>()?)
30                }
31                other => return Err(Error::new_spanned(other, "expected `invariant(...)`")),
32            }
33        }
34    }
35    Ok(invariants)
36}

38/// Validates invariant syntax and placement, returning the first invariant's
39/// span so derives which cannot honor invariants can reject them.
40pub(crate) fn validate(ast: &DeriveInput) -> Result<Option<Span>, Error> {
41    let fields: Vec<&Field> = match &ast.data {
42        Data::Struct(data) => data.fields.iter().collect(),
43        Data::Enum(data) => {
44            for variant in &data.variants {
45                if let Some(invariant) = parse(&variant.attrs)?.first() {
46                    return Err(Error::new(
47                        invariant.span(),
48                        "invariants are only supported on named fields",
49                    ));
50                }
51            }
52            data.variants.iter().flat_map(|variant| &variant.fields).collect()
53        }
54        Data::Union(data) => data.fields.named.iter().collect(),
55    };
56    let mut first = None;
57    for field in fields {
58        if let Some(invariant) = parse(&field.attrs)?.first() {
59            if field.ident.is_none() {
60                return Err(Error::new(
61                    invariant.span(),
62                    "invariants are only supported on named fields",
63                ));
64            }
65            first.get_or_insert(invariant.span());
66        }
67    }
68    Ok(first)
69}

71/// Visits identifiers, including those inside macro arguments.
72pub(crate) fn idents(tokens: TokenStream) -> Vec<Ident> {
73    tokens
74        .into_iter()
75        .flat_map(|token| match token {
76            TokenTree::Ident(ident) => vec![ident],
77            TokenTree::Group(group) => idents(group.stream()),
78            _ => Vec::new(),
79        })
80        .collect()
81}

83/// Binds semantic field names in every syntax context used by the expression.
84/// Preserve the expression's tokens so its other names retain their hygiene.
85pub(crate) fn bind_fields(fields: &[(&Field, Ident)], expression: &Expr) -> TokenStream {
86    let mut body = expression.to_token_stream();
87    let references = idents(body.clone());
88    for (field, binding) in fields {
89        let name = match &field.ident {
90            Some(name) => name,
91            None => continue,
92        };
93        let spelling = name.to_string();
94        let spelling = spelling.trim_start_matches("r#");
95        // Also support references produced by a macro invoked by the expression.
96        let mut call_site = name.clone();
97        call_site.set_span(Span::call_site());
98        for alias in [name.clone(), call_site].into_iter().chain(
99            references
100                .iter()
101                .filter(|ident| ident.to_string().trim_start_matches("r#") == spelling)
102                .cloned(),
103        ) {
104            body = quote! {{
105                // Require a binding; reject names Rust would resolve as constant
106                // patterns. Item aliases could capture helper calls in macros.
107                #[allow(unused_variables, non_snake_case, clippy::redundant_pattern)]
108                let #alias @ _ = #binding;
109                #body
110            }};
111        }
112    }
113    body
114}
