25/// The computed representation of a type.
26///
27/// This is the result of processing all `#[repr(...)]` attributes on a type, if
28/// any. A `Repr` is only capable of representing legal combinations of
29/// `#[repr(...)]` attributes.
30#[cfg_attr(test, derive(Copy, Clone, Debug))]
31pub(crate) enum Repr<Prim, Packed> {
32    /// `#[repr(transparent)]`
33    Transparent(Span),
34    /// A compound representation: `repr(C)`, `repr(Rust)`, or `repr(Int)`
35    /// optionally combined with `repr(packed(...))` or `repr(align(...))`
36    Compound(Spanned<CompoundRepr<Prim>>, Option<Spanned<AlignRepr<Packed>>>),
37}
38
39/// A compound representation: `repr(C)`, `repr(Rust)`, or `repr(Int)`.
40#[cfg_attr(test, derive(Copy, Clone, Debug, Eq, PartialEq))]
41pub(crate) enum CompoundRepr<Prim> {
42    C,
43    Rust,
44    Primitive(Prim),
45}

47/// `repr(Int)`
48#[derive(Copy, Clone)]
49#[cfg_attr(test, derive(Debug, Eq, PartialEq))]
50pub(crate) enum PrimitiveRepr {
51    U8,
52    U16,
53    U32,
54    U64,
55    U128,
56    Usize,
57    I8,
58    I16,
59    I32,
60    I64,
61    I128,
62    Isize,
63}

66#[cfg_attr(test, derive(Copy, Clone, Debug, Eq, PartialEq))]
67pub(crate) enum AlignRepr<Packed> {
68    Packed(Packed),

78impl<Prim, Packed> Repr<Prim, Packed> {
79    /// Gets the name of this "repr type" - the non-align `repr(X)` that is used
80    /// in prose to refer to this type.
81    ///
82    /// For example, we would refer to `#[repr(C, align(4))] struct Foo { ... }`
83    /// as a "`repr(C)` struct".
84    pub(crate) fn repr_type_name(&self) -> &str
85    where
86        Prim: Copy + With<PrimitiveRepr>,
87    {
88        use CompoundRepr::*;
89        use PrimitiveRepr::*;
90        use Repr::*;
91        match self {
92            Transparent(_span) => "repr(transparent)",
93            Compound(Spanned { t: repr, span: _ }, _align) => match repr {
94                C => "repr(C)",
95                Rust => "repr(Rust)",
96                Primitive(prim) => prim.with(|prim| match prim {
97                    U8 => "repr(u8)",
98                    U16 => "repr(u16)",
99                    U32 => "repr(u32)",
100                    U64 => "repr(u64)",
101                    U128 => "repr(u128)",
102                    Usize => "repr(usize)",
103                    I8 => "repr(i8)",
104                    I16 => "repr(i16)",
105                    I32 => "repr(i32)",
106                    I64 => "repr(i64)",
107                    I128 => "repr(i128)",
108                    Isize => "repr(isize)",
109                }),
110            },
111        }
112    }
113
114    pub(crate) fn is_transparent(&self) -> bool {
115        matches!(self, Repr::Transparent(_))
116    }
117
118    pub(crate) fn is_c(&self) -> bool {
119        use CompoundRepr::*;
120        matches!(self, Repr::Compound(Spanned { t: C, span: _ }, _align))
121    }
122
123    pub(crate) fn is_primitive(&self) -> bool {
124        use CompoundRepr::*;
125        matches!(self, Repr::Compound(Spanned { t: Primitive(_), span: _ }, _align))
126    }
127
128    pub(crate) fn get_packed(&self) -> Option<&Packed> {
129        use AlignRepr::*;
130        use Repr::*;
131        if let Compound(_, Some(Spanned { t: Packed(p), span: _ })) = self {
132            Some(p)
133        } else {
134            None
135        }
136    }
137
138    pub(crate) fn get_align(&self) -> Option<Spanned<NonZeroU32>> {
139        use AlignRepr::*;
140        use Repr::*;
141        if let Compound(_, Some(Spanned { t: Align(n), span })) = self {
142            Some(Spanned::new(*n, *span))
143        } else {
144            None
145        }
146    }
147
148    pub(crate) fn is_align_gt_1(&self) -> bool {
149        self.get_align().map(|n| n.t.get() > 1).unwrap_or(false)
150    }
151
152    /// When deriving `Unaligned`, validate that the decorated type has no
153    /// `#[repr(align(N))]` attribute where `N > 1`. If no such attribute exists
154    /// (including if `N == 1`), this returns `Ok(())`, and otherwise it returns
155    /// a descriptive error.
156    pub(crate) fn unaligned_validate_no_align_gt_1(&self) -> Result<(), Error> {
157        if let Some(n) = self.get_align().filter(|n| n.t.get() > 1) {
158            Err(Error::new(
159                n.span,
160                "cannot derive `Unaligned` on type with alignment greater than 1",
161            ))
162        } else {
163            Ok(())
164        }
165    }
166}

197impl<Prim, Packed> ToTokens for Repr<Prim, Packed>
198where
199    Prim: With<PrimitiveRepr> + Copy,
200    Packed: With<NonZeroU32> + Copy,
201{
202    fn to_tokens(&self, ts: &mut TokenStream) {
203        use Repr::*;
204        match self {
205            Transparent(span) => ts.append_all(quote_spanned! { *span=> #[repr(transparent)] }),
206            Compound(repr, align) => {
207                repr.to_tokens(ts);
208                if let Some(align) = align {
209                    align.to_tokens(ts);
210                }
211            }
212        }
213    }
214}

216impl<Prim: With<PrimitiveRepr> + Copy> ToTokens for Spanned<CompoundRepr<Prim>> {

218        use CompoundRepr::*;

221            Rust => ts.append_all(quote_spanned! { self.span=> #[repr(Rust)] }),
222            Primitive(prim) => prim.with(|prim| Spanned::new(prim, self.span).to_tokens(ts)),
223        }
224    }

247impl<Packed: With<NonZeroU32> + Copy> ToTokens for Spanned<AlignRepr<Packed>> {
248    fn to_tokens(&self, ts: &mut TokenStream) {
249        use AlignRepr::*;
250        // We use `syn::Index` instead of `u32` because `quote_spanned!`
251        // serializes `u32` literals as `123u32`, not just `123`. Rust doesn't
252        // recognize that as a valid argument to `#[repr(align(...))]` or
253        // `#[repr(packed(...))]`.
254        let to_index = |n: NonZeroU32| syn::Index { index: n.get(), span: self.span };
255        match self.t {
256            Packed(n) => n.with(|n| {
257                let n = to_index(n);
258                ts.append_all(quote_spanned! { self.span => #[repr(packed(#n))] })
259            }),
260            Align(n) => {
261                let n = to_index(n);
262                ts.append_all(quote_spanned! { self.span => #[repr(align(#n))] })
263            }
264        }
265    }
266}

293/// The error from converting from a `RawRepr`.
294#[cfg_attr(test, derive(Debug, Eq, PartialEq))]
295pub(crate) enum FromRawReprError<E> {
296    /// The `RawRepr` doesn't affect the high-level repr we're parsing (e.g.
297    /// it's `align(...)` and we're parsing a `CompoundRepr`).
298    None,
299    /// The `RawRepr` is invalid for the high-level repr we're parsing (e.g.
300    /// it's `packed` repr and we're parsing an `AlignRepr` for an enum type).
301    Err(E),
302}
303
304/// The representation hint is not supported for the decorated type.
305#[cfg_attr(test, derive(Copy, Clone, Debug, Eq, PartialEq))]
306pub(crate) struct UnsupportedReprError;

308impl<Prim: With<PrimitiveRepr>> TryFrom<RawRepr> for CompoundRepr<Prim> {
309    type Error = FromRawReprError<UnsupportedReprError>;
310    fn try_from(
311        raw: RawRepr,
312    ) -> Result<CompoundRepr<Prim>, FromRawReprError<UnsupportedReprError>> {
313        use RawRepr::*;
314        match raw {
315            C => Ok(CompoundRepr::C),
316            Rust => Ok(CompoundRepr::Rust),
317            raw @ (U8 | U16 | U32 | U64 | U128 | Usize | I8 | I16 | I32 | I64 | I128 | Isize) => {
318                Prim::try_with_or(
319                    || match raw {
320                        U8 => Ok(PrimitiveRepr::U8),
321                        U16 => Ok(PrimitiveRepr::U16),
322                        U32 => Ok(PrimitiveRepr::U32),
323                        U64 => Ok(PrimitiveRepr::U64),
324                        U128 => Ok(PrimitiveRepr::U128),
325                        Usize => Ok(PrimitiveRepr::Usize),
326                        I8 => Ok(PrimitiveRepr::I8),
327                        I16 => Ok(PrimitiveRepr::I16),
328                        I32 => Ok(PrimitiveRepr::I32),
329                        I64 => Ok(PrimitiveRepr::I64),
330                        I128 => Ok(PrimitiveRepr::I128),
331                        Isize => Ok(PrimitiveRepr::Isize),
332                        Transparent | C | Rust | Align(_) | PackedN(_) | Packed => {
333                            Err(UnsupportedReprError)
334                        }
335                    },
336                    UnsupportedReprError,
337                )
338                .map(CompoundRepr::Primitive)
339                .map_err(FromRawReprError::Err)
340            }
341            Transparent | Align(_) | PackedN(_) | Packed => Err(FromRawReprError::None),
342        }
343    }
344}

346impl<Pcked: With<NonZeroU32>> TryFrom<RawRepr> for AlignRepr<Pcked> {
347    type Error = FromRawReprError<UnsupportedReprError>;
348    fn try_from(raw: RawRepr) -> Result<AlignRepr<Pcked>, FromRawReprError<UnsupportedReprError>> {
349        use RawRepr::*;
350        match raw {
351            Packed | PackedN(_) => Pcked::try_with_or(
352                || match raw {
353                    Packed => Ok(NonZeroU32::new(1).unwrap()),
354                    PackedN(n) => Ok(n),
355                    U8 | U16 | U32 | U64 | U128 | Usize | I8 | I16 | I32 | I64 | I128 | Isize
356                    | Transparent | C | Rust | Align(_) => Err(UnsupportedReprError),
357                },
358                UnsupportedReprError,
359            )
360            .map(AlignRepr::Packed)
361            .map_err(FromRawReprError::Err),
362            Align(n) => Ok(AlignRepr::Align(n)),
363            U8 | U16 | U32 | U64 | U128 | Usize | I8 | I16 | I32 | I64 | I128 | Isize
364            | Transparent | C | Rust => Err(FromRawReprError::None),
365        }
366    }
367}

369/// The error from extracting a high-level repr type from a list of `RawRepr`s.
370#[cfg_attr(test, derive(Copy, Clone, Debug, Eq, PartialEq))]
371enum FromRawReprsError<E> {
372    /// One of the `RawRepr`s is invalid for the high-level repr we're parsing
373    /// (e.g. there's a `packed` repr and we're parsing an `AlignRepr` for an
374    /// enum type).
375    Single(E),
376    /// Two `RawRepr`s appear which both affect the high-level repr we're
377    /// parsing (e.g., the list is `#[repr(align(2), packed)]`). Note that we
378    /// conservatively treat redundant reprs as conflicting (e.g.
379    /// `#[repr(packed, packed)]`).
380    Conflict,
381}
382
383/// Tries to extract a high-level repr from a list of `RawRepr`s.
384fn try_from_raw_reprs<'a, E, R: TryFrom<RawRepr, Error = FromRawReprError<E>>>(
385    r: impl IntoIterator<Item = &'a Spanned<RawRepr>>,
386) -> Result<Option<Spanned<R>>, Spanned<FromRawReprsError<E>>> {
387    // Walk the list of `RawRepr`s and attempt to convert each to an `R`. Bail
388    // if we find any errors. If we find more than one which converts to an `R`,
389    // bail with a `Conflict` error.
390    r.into_iter().try_fold(None, |found: Option<Spanned<R>>, raw| {
391        let new = match Spanned::<R>::try_from(*raw) {
392            Ok(r) => r,
393            // This `RawRepr` doesn't convert to an `R`, so keep the current
394            // found `R`, if any.
395            Err(FromRawReprError::None) => return Ok(found),
396            // This repr is unsupported for the decorated type (e.g.
397            // `repr(packed)` on an enum).
398            Err(FromRawReprError::Err(Spanned { t: err, span })) => {
399                return Err(Spanned::new(FromRawReprsError::Single(err), span))
400            }
401        };
402
403        if let Some(found) = found {
404            // We already found an `R`, but this `RawRepr` also converts to an
405            // `R`, so that's a conflict.
406            //
407            // `Span::join` returns `None` if the two spans are from different
408            // files or if we're not on the nightly compiler. In that case, just
409            // use `new`'s span.
410            let span = found.span.join(new.span).unwrap_or(new.span);
411            Err(Spanned::new(FromRawReprsError::Conflict, span))
412        } else {
413            Ok(Some(new))
414        }
415    })
416}

418/// The error returned from [`Repr::from_attrs`].
419#[cfg_attr(test, derive(Copy, Clone, Debug, Eq, PartialEq))]
420enum FromAttrsError {
421    FromRawReprs(FromRawReprsError<UnsupportedReprError>),
422    Unrecognized,
423}
424
425impl From<FromRawReprsError<UnsupportedReprError>> for FromAttrsError {
426    fn from(err: FromRawReprsError<UnsupportedReprError>) -> FromAttrsError {
427        FromAttrsError::FromRawReprs(err)
428    }
429}
430
431impl From<UnrecognizedReprError> for FromAttrsError {
432    fn from(_err: UnrecognizedReprError) -> FromAttrsError {
433        FromAttrsError::Unrecognized
434    }
435}
436
437impl From<Spanned<FromAttrsError>> for Error {
438    fn from(err: Spanned<FromAttrsError>) -> Error {
439        let Spanned { t: err, span } = err;
440        match err {
441            FromAttrsError::FromRawReprs(FromRawReprsError::Single(
442                _err @ UnsupportedReprError,
443            )) => Error::new(span, "unsupported representation hint for the decorated type"),
444            FromAttrsError::FromRawReprs(FromRawReprsError::Conflict) => {
445                // NOTE: This says "another" rather than "a preceding" because
446                // when one of the reprs involved is `transparent`, we detect
447                // that condition in `Repr::from_attrs`, and at that point we
448                // can't tell which repr came first, so we might report this on
449                // the first involved repr rather than the second, third, etc.
450                Error::new(span, "this conflicts with another representation hint")
451            }
452            FromAttrsError::Unrecognized => Error::new(span, "unrecognized representation hint"),
453        }
454    }
455}

457impl<Prim, Packed> Repr<Prim, Packed> {
458    fn from_attrs_inner(attrs: &[Attribute]) -> Result<Repr<Prim, Packed>, Spanned<FromAttrsError>>
459    where
460        Prim: With<PrimitiveRepr>,
461        Packed: With<NonZeroU32>,
462    {
463        let raw_reprs = RawRepr::from_attrs(attrs).map_err(Spanned::from)?;
464
465        let transparent = {
466            let mut transparents = raw_reprs.iter().filter_map(|Spanned { t, span }| match t {
467                RawRepr::Transparent => Some(span),
468                _ => None,
469            });
470            let first = transparents.next();
471            let second = transparents.next();
472            match (first, second) {
473                (None, None) => None,
474                (Some(span), None) => Some(*span),
475                (Some(_), Some(second)) => {
476                    return Err(Spanned::new(
477                        FromAttrsError::FromRawReprs(FromRawReprsError::Conflict),
478                        *second,
479                    ))
480                }
481                // An iterator can't produce a value only on the second call to
482                // `.next()`.
483                (None, Some(_)) => unreachable!(),
484            }
485        };
486
487        let compound: Option<Spanned<CompoundRepr<Prim>>> =
488            try_from_raw_reprs(raw_reprs.iter()).map_err(Spanned::from)?;
489        let align: Option<Spanned<AlignRepr<Packed>>> =
490            try_from_raw_reprs(raw_reprs.iter()).map_err(Spanned::from)?;
491
492        if let Some(span) = transparent {
493            if compound.is_some() || align.is_some() {
494                // Arbitrarily report the problem on the `transparent` span. Any
495                // span will do.
496                return Err(Spanned::new(FromRawReprsError::Conflict.into(), span));
497            }
498
499            Ok(Repr::Transparent(span))
500        } else {
501            Ok(Repr::Compound(
502                compound.unwrap_or(Spanned::new(CompoundRepr::Rust, Span::call_site())),
503                align,
504            ))
505        }
506    }
507}

519/// The representation hint could not be parsed or was unrecognized.
520struct UnrecognizedReprError;
521
522impl RawRepr {
523    fn from_attrs(
524        attrs: &[Attribute],
525    ) -> Result<Vec<Spanned<RawRepr>>, Spanned<UnrecognizedReprError>> {
526        let mut reprs = Vec::new();
527        for attr in attrs {
528            // Ignore documentation attributes.
529            if path_is_ident(attr.path(), "doc") {
530                continue;
531            }
532            if let Meta::List(ref meta_list) = attr.meta {
533                if path_is_ident(&meta_list.path, "repr") {
534                    let parsed: Punctuated<Meta, Comma> =
535                        match meta_list.parse_args_with(Punctuated::parse_terminated) {
536                            Ok(parsed) => parsed,
537                            Err(_) => {
538                                return Err(Spanned::new(
539                                    UnrecognizedReprError,
540                                    meta_list.tokens.span(),
541                                ))
542                            }
543                        };
544                    for meta in parsed {
545                        let s = meta.span();
546                        reprs.push(
547                            RawRepr::from_meta(&meta)
548                                .map(|r| Spanned::new(r, s))
549                                .map_err(|e| Spanned::new(e, s))?,
550                        );
551                    }
552                }
553            }
554        }
555
556        Ok(reprs)
557    }
558
559    fn from_meta(meta: &Meta) -> Result<RawRepr, UnrecognizedReprError> {
560        let (path, list) = match meta {
561            Meta::Path(path) => (path, None),
562            Meta::List(list) => (&list.path, Some(list)),
563            _ => return Err(UnrecognizedReprError),
564        };
565
566        let ident = path.get_ident().ok_or(UnrecognizedReprError)?;
567
568        // Only returns `Ok` for non-zero power-of-two values.
569        let parse_nzu64 = |list: &MetaList| {
570            list.parse_args::<LitInt>()
571                .and_then(|int| int.base10_parse::<NonZeroU32>())
572                .map_err(|_| UnrecognizedReprError)
573                .and_then(|nz| {
574                    if nz.get().is_power_of_two() {
575                        Ok(nz)
576                    } else {
577                        Err(UnrecognizedReprError)
578                    }
579                })
580        };
581
582        use RawRepr::*;
583        let ident = to_ident_str(ident);
584        Ok(match (ident.as_str(), list) {
585            ("u8", None) => U8,
586            ("u16", None) => U16,
587            ("u32", None) => U32,
588            ("u64", None) => U64,
589            ("u128", None) => U128,
590            ("usize", None) => Usize,
591            ("i8", None) => I8,
592            ("i16", None) => I16,
593            ("i32", None) => I32,
594            ("i64", None) => I64,
595            ("i128", None) => I128,
596            ("isize", None) => Isize,
597            ("C", None) => C,
598            ("transparent", None) => Transparent,
599            ("Rust", None) => Rust,
600            ("packed", None) => Packed,
601            ("packed", Some(list)) => PackedN(parse_nzu64(list)?),
602            ("align", Some(list)) => Align(parse_nzu64(list)?),
603            _ => return Err(UnrecognizedReprError),
604        })
605    }

609mod util {
610    use super::*;
611    /// A value with an associated span.
612    #[derive(Copy, Clone)]
613    #[cfg_attr(test, derive(Debug))]
614    pub(crate) struct Spanned<T> {
615        pub(crate) t: T,
616        pub(crate) span: Span,
617    }
618
619    impl<T> Spanned<T> {
620        pub(super) fn new(t: T, span: Span) -> Spanned<T> {
621            Spanned { t, span }
622        }
623
624        pub(super) fn from<U>(s: Spanned<U>) -> Spanned<T>
625        where
626            T: From<U>,
627        {
628            let Spanned { t: u, span } = s;
629            Spanned::new(u.into(), span)
630        }
631
632        /// Delegates to `T: TryFrom`, preserving span information in both the
633        /// success and error cases.
634        pub(super) fn try_from<E, U>(
635            u: Spanned<U>,
636        ) -> Result<Spanned<T>, FromRawReprError<Spanned<E>>>
637        where
638            T: TryFrom<U, Error = FromRawReprError<E>>,
639        {
640            let Spanned { t: u, span } = u;
641            T::try_from(u).map(|t| Spanned { t, span }).map_err(|err| match err {
642                FromRawReprError::None => FromRawReprError::None,
643                FromRawReprError::Err(e) => FromRawReprError::Err(Spanned::new(e, span)),
644            })
645        }
646    }
647
648    // Used to permit implementing `With<T> for T: Inhabited` and for
649    // `Infallible` without a blanket impl conflict.
650    pub(crate) trait Inhabited {}
651    impl Inhabited for PrimitiveRepr {}
652    impl Inhabited for NonZeroU32 {}
653
654    pub(crate) trait With<T> {
655        fn with<O, F: FnOnce(T) -> O>(self, f: F) -> O;
656        fn try_with_or<E, F: FnOnce() -> Result<T, E>>(f: F, err: E) -> Result<Self, E>
657        where
658            Self: Sized;
659    }
660
661    impl<T: Inhabited> With<T> for T {
662        fn with<O, F: FnOnce(T) -> O>(self, f: F) -> O {
663            f(self)
664        }
665
666        fn try_with_or<E, F: FnOnce() -> Result<T, E>>(f: F, _err: E) -> Result<Self, E> {
667            f()
668        }
669    }
670
671    impl<T> With<T> for Infallible {
672        fn with<O, F: FnOnce(T) -> O>(self, _f: F) -> O {
673            match self {}
674        }
675
676        fn try_with_or<E, F: FnOnce() -> Result<T, E>>(_f: F, err: E) -> Result<Self, E> {
677            Err(err)
678        }
679    }
680}
