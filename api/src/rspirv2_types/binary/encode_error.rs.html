1use std::error::Error;
2use std::fmt::{Debug, Display, Formatter};
3
4#[derive(Clone, PartialEq)]
5pub enum EncodeError {
6    /// A customizable error
7    CustomError(String),
8    OpTooLong,
9    OutOfIdResults,
10}
11
12impl Display for EncodeError {
13    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
14        match self {
15            EncodeError::CustomError(err) => write!(f, "{}", err),
16            EncodeError::OpTooLong => write!(f, "Op too long, u16 overflow"),
17            EncodeError::OutOfIdResults => write!(f, "Ran out of IdResults, danger of overflow!"),
18        }
19    }
20}
21
22impl Debug for EncodeError {
23    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
24        Display::fmt(self, f)
25    }
26}
27
28impl Error for EncodeError {}
