6/// A `WordWriter` allows you to [`Self::write`] [`Word`]s into it and is usually backed by a [`Vec`].
7///
8/// Implementations include `Vec<Word>` and [`WordCounter`].
9pub trait WordWriter: Sized {
10    /// Write a single [`Word`]
11    fn write(&mut self, word: Word);
12
13    /// Write an [`Iterator`] of [`Word`]s
14    fn write_iter(&mut self, iter: impl IntoIterator<Item = Word>);
15
16    /// Encode and write the opcode and len of an instruction.
17    ///
18    /// Should not be overwritten.
19    #[inline]
20    fn write_op(&mut self, op: u16, len: usize) -> Result<(), EncodeError> {
21        self.inst_reserve(len);
22        self.write(Word::new_op(op, len)?);
23        Ok(())
24    }
25
26    /// Expect this many words to be written by the next instruction.
27    ///
28    /// Usually called right before an instruction is emitted, with the length of the instruction, so expect this to be
29    /// called quite often. If you're writing into a [`Vec`], prefer [`Vec::reserve`] over [`Vec::reserve_exact`].
30    #[inline]
31    fn inst_reserve(&mut self, len: usize) {
32        let _ = len;
33    }
34}

36impl WordWriter for Vec<Word> {
37    #[inline]
38    fn write(&mut self, word: Word) {
39        self.push(word);
40    }
41
42    #[inline]
43    fn write_iter(&mut self, iter: impl IntoIterator<Item = Word>) {
44        self.extend(iter);
45    }
46
47    #[inline]
48    fn inst_reserve(&mut self, len: usize) {
49        self.reserve(len);
50    }
51}

87/// Counts the amount of [`Word`]s that were written, discarding the words itself
88#[derive(Clone, Debug, Default)]
89pub struct WordCounter(pub usize);
90
91impl Deref for WordCounter {
92    type Target = usize;
93
94    #[inline]
95    fn deref(&self) -> &Self::Target {
96        &self.0
97    }
98}
99
100impl DerefMut for WordCounter {
101    #[inline]
102    fn deref_mut(&mut self) -> &mut Self::Target {
103        &mut self.0
104    }
105}
106
107impl WordWriter for WordCounter {
108    #[inline]
109    fn write(&mut self, _: Word) {
110        self.0 += 1;
111    }
112
113    #[inline]
114    fn write_iter(&mut self, iter: impl IntoIterator<Item = Word>) {
115        self.0 += iter.into_iter().count();
116    }
117}

119pub struct WordSliceWriter<'a> {
120    slice: &'a mut [Word],
121    offset: usize,
122}
123
124impl<'a> WordSliceWriter<'a> {
125    pub fn new(slice: &'a mut [Word]) -> Self {
126        Self { slice, offset: 0 }
127    }
128
129    pub fn finalize(self) {
130        assert_eq!(
131            self.offset,
132            self.slice.len(),
133            "Wrote {} Words but expected {} Words to be written",
134            self.offset,
135            self.slice.len()
136        );
137    }
138}
139
140impl WordWriter for WordSliceWriter<'_> {
141    fn write(&mut self, word: Word) {
142        self.slice[self.offset] = word;
143        self.offset += 1;
144    }
145
146    fn write_iter(&mut self, iter: impl IntoIterator<Item = Word>) {
147        for word in iter {
148            self.write(word);
149        }
150    }
151}
