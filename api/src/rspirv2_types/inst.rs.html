14pub trait InstEncoding: Sized + Debug + Eq {
15    /// `IdResult` is either an [`IdResult`] or `()`, depending on whether this Instruction has an [`IdResult`].
16    type IdResult: MaybeIdResult;
17    /// `IdResult` is either an [`IdResult`] or `()`, depending on whether this Instruction has an
18    /// [`crate::operand::IdResultType`].
19    type IdResultType: MaybeIdResult;
20
21    /// Query the potential [`IdResult`] of this Instruction, or `()` if it has none.
22    fn id_result(&self) -> Self::IdResult;
23
24    /// Query the potential [`IdResult`] of this Instruction, or `()` if it has none.
25    fn id_result_type(&self) -> Self::IdResultType;
26
27    /// Name of the instruction set, for debug printing
28    fn name() -> &'static str {
29        "unknown"
30    }
31
32    /// Encode this instruction to a [`WordWriter`]
33    fn encode(&self, writer: &mut impl WordWriter) -> Result<(), EncodeError>;
34
35    /// Decode this instruction from an [`InstReader`], error when opcode is unknown.
36    ///
37    /// See [`Self::try_decode`] for a variant that returns `None` when opcode is unknown.
38    fn decode(reader: InstReader<'_>) -> Result<Self, DecodeError>;
39
40    /// Try to decode this instruction from an [`InstReader`], return `None` when the opcode is unknown.
41    ///
42    /// See [`Self::decode`] for a variant that errors when opcode is unknown.
43    #[inline]
44    fn try_decode(reader: InstReader<'_>) -> Result<Option<Self>, DecodeError> {
45        match Self::decode(reader) {
46            Ok(e) => Ok(Some(e)),
47            Err(DecodeError {
48                kind: DecodeErrorKind::WrongOpCode { .. } | DecodeErrorKind::UnknownOpCode { .. },
49                ..
50            }) => Ok(None),
51            Err(e) => Err(e),
52        }
53    }
54
55    /// Disassemble this instruction to the supplied [`Formatter`] `f`
56    fn dis_fmt(&self, f: &mut Formatter<'_>, ctx: &DisContext) -> std::fmt::Result;
57
58    /// Disassemble this instruction, returns a type that impl [`Display`] you may use in `format!`.
59    ///
60    /// You shouldn't overwrite this function but [`Self::dis_fmt`] instead.
61    #[inline]
62    fn dis<'a>(&'a self, ctx: &'a DisContext) -> InstDis<'a, Self> {
63        InstDis(self, ctx)
64    }
65}

126/// A reference to a valid instruction encoded in a slice of [`Word`]s.
127///
128/// Call [`Self::get`] to get the underlying instruction, as it unfortunately can't implement [`Deref`].
129///
130/// # Safety
131/// The referenced words must be an encoded instruction that is valid within the `ISA` instruction set. Encountering an
132/// invalid instruction may panic.
133pub struct InstRef<'a, ISA: InstEncoding> {
134    reader: InstReader<'a>,
135    _phantom: PhantomData<ISA>,
136}
137
138impl<'a, ISA: InstEncoding> InstRef<'a, ISA> {
139    #[inline]
140    pub fn from_words_unchecked(words: &'a [Word]) -> Result<Self, DecodeError> {
141        Ok(Self {
142            reader: InstReader::from_words(words)?,
143            _phantom: PhantomData,
144        })
145    }
146
147    #[inline]
148    pub fn get(&self) -> ISA {
149        ISA::decode(**self).unwrap()
150    }
151}
152
153impl<'a, ISA: InstEncoding> Deref for InstRef<'a, ISA> {
154    type Target = InstReader<'a>;
155
156    #[inline]
157    fn deref(&self) -> &Self::Target {
158        &self.reader
159    }
160}

162impl<ISA: InstEncoding> Copy for InstRef<'_, ISA> {}

167    }
