15/// Options for disassembly
16#[derive(Clone, Debug)]
17pub struct DisOptions {
18    /// Whether to emit ANSI escape sequences for colorful output
19    pub color: bool,
20    /// Describes how to escape string sequences
21    pub literal_string_escape: LiteralStringEscape,
22    /// Add extra spaces around [`crate::operand::IdResultType`] to match `rspirv`'s behaviour
23    pub rspirv_space: bool,
24    /// Format constants based on the `IdResultType` of the `OpConstant`, requires scanning the module for type
25    /// information to provide context.
26    ///
27    /// **REQUIRED** for correct disassembly
28    pub const_fmt: bool,
29    /// Defines whether to use raw ids (`%53`) or named ids (`%my_name`) resolved from `OpName` descriptors or other
30    /// sources of metadata. See [`IdNaming`].
31    pub id_naming: IdNaming,
32    /// How to derive names for types
33    pub type_naming: TypeNaming,
34    /// The padding to add before each instruction to make the `=` be on the same line. Usually N-many spaces.
35    pub padding: String,
36}
37
38/// Defines where names of types should be derived from.
39#[derive(Copy, Clone, Debug, Eq, PartialEq)]
40pub enum IdNaming {
41    /// Only use raw ids (`%53`)
42    RawId,
43    /// Only use names for explicitly named types (`OpName`)
44    ExplicitNames,
45    /// Use names from both explicit naming (`OpName`) and other sources of metadata.
46    ///
47    /// Typical sources of names, besides `OpName`:
48    /// * `OpType*`: `%u32 = OpTypeInt 32 0`
49    /// * `OpConstant`: `%u32_42 = OpConstant %u32 42`
50    All,
51}
52
53/// Defines how to derive names for types
54#[derive(Copy, Clone, Debug, Eq, PartialEq)]
55pub enum TypeNaming {
56    /// Rust naming conventions, eg. `u32`, `i32`, `f32`
57    Rust,
58    /// C naming conventions, eg. `uint`, `int`, `float`
59    ///
60    /// should match spirv-dis
61    C,
62}
63
64/// `&str` of 15 spaces, the default padding in spirv-tools
65pub const STR_15_SPACES: &str = "               ";

134/// Context object for disassembly generation
135///
136/// Next to [`DisOptions`], contains lookup tables to aid in disassembly generation. Initializing these tables is
137/// **required** to generate correct disassembly that `spirv-as` can read, otherwise, we'll generate the best possible
138/// disassembly we can. Since instructions are not available within the `rspirv2-types` crate, we can't actually
139/// implement the context information retrieval here. Instead, it needs to be implemented for each instruction set with
140/// the [`InstSetDisCtx`] trait, which may delegate to our default implementation in
141/// `rspirv2::dis::create_dis_context_core`.
142#[derive(Clone, Debug, Default)]
143pub struct DisContext {
144    /// Options
145    opt: DisOptions,
146    /// Maps an [`IdResult`] of an arbitrary instruction to its integer width, if it is an integer.
147    /// To decode an `OpSwitch`, one needs to know from context whether the selector is 32 or 64bit, as it determines
148    /// whether the label constants consume one or two words.
149    pub id_to_int_width: FxHashMap<IdResult, u32>,
150    /// Maps an [`IdResult`] of a type declaration to a [`PrimitiveType`]. Primarily for `OpSwitch`, but such a lookup
151    /// may be useful for other instructions as well.
152    pub id_type_to_primitive_type: FxHashMap<IdResult, PrimitiveType>,
153    /// Maps an [`IdResult`] of a type declaration to a [`ConstFmt`] to tell `OpConstant` instructions how to format
154    /// the untyped constant value
155    pub id_to_const_fmt: FxHashMap<IdResult, ConstFmt>,
156    /// Maps an [`IdResult`] to the "highest priority" [`IdName`]
157    pub id_to_name: FxHashMap<IdResult, IdName>,
158}
159
160/// An enum for representing primitive types
161#[derive(Copy, Clone, Debug)]
162pub enum PrimitiveType {
163    Void,
164    Bool,
165    Int { width: u32, signedness: bool },
166    Float { width: u32 },
167}

169#[derive(Clone, Debug, Default)]
170pub enum IdName {
171    /// Use the raw id
172    #[default]
173    RawId,
174    /// Use this explicitly assigned name, has priority over [`Self::DerivedName`]
175    ExplicitName(String),
176    /// Derived name from the definition of the id
177    DerivedName(String),
178}
179
180#[derive(Copy, Clone, Debug)]
181pub enum ResolvedIdName<'a> {
182    Named(&'a str),
183    Raw(IdResult),
184}
185
186impl Display for ResolvedIdName<'_> {
187    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
188        match self {
189            ResolvedIdName::Named(name) => write!(f, "{name}"),
190            ResolvedIdName::Raw(id) => write!(f, "{}", id.0.0),
191        }
192    }
193}
194
195impl DisContext {
196    /// Creates a new [`DisContext`] without having scanned the module for the required extra information.
197    ///
198    /// **WARNING**: You need to [`Self::add_context`] the instructions you want to decode before disassembling,
199    /// as the disassembly generated may be invalid without the required context. For example, `OpConstant` needs to
200    /// query the type of constant their value is to format it correctly as a float or an int.
201    #[inline]
202    pub fn no_context(opt: DisOptions) -> Self {
203        Self {
204            opt,
205            ..Default::default()
206        }
207    }
208
209    /// Scan the supplied [`InstSlice`] for useful context
210    pub fn add_context<ISA: InstSetDisCtx>(&mut self, slice: &InstSlice<ISA>) {
211        profiling::function_scope!();
212        ISA::add_context(slice.as_raw(), self);
213    }
214
215    /// Scan the supplied [`InstSlice`] for useful context, skip over any [`DecodeError`]s that may arise
216    ///
217    /// [`DecodeError`]: crate::binary::DecodeError
218    pub fn add_context_raw<ISA: InstSetDisCtx>(&mut self, slice: &RawInstSlice) {
219        profiling::function_scope!();
220        ISA::add_context(slice, self);
221    }
222
223    /// Add a mapping from an [`IdResult`] to an [`IdName`]
224    pub fn add_id_to_name(&mut self, id: IdResult, name: IdName) {
225        // In the name of Hades, I accept this message!
226        #[allow(clippy::match_same_arms)]
227        let accept = match (self.opt.id_naming, &name) {
228            // no need to store anything, default is `RawId`
229            (_, IdName::RawId) => false,
230            (IdNaming::RawId, _) => false,
231
232            (IdNaming::All, _) => true,
233            (IdNaming::ExplicitNames, IdName::ExplicitName(..)) => true,
234            (IdNaming::ExplicitNames, IdName::DerivedName(..)) => false,
235        };
236        if accept {
237            match self.id_to_name.entry(id) {
238                Entry::Occupied(mut slot) => {
239                    let overwrite = match (&name, slot.get()) {
240                        // don't overwrite an explicit name
241                        (IdName::DerivedName(..), IdName::ExplicitName(..)) => false,
242                        (_, _) => true,
243                    };
244                    if overwrite {
245                        slot.insert(name);
246                    }
247                }
248                Entry::Vacant(slot) => {
249                    slot.insert(name);
250                }
251            }
252        }
253    }
254
255    pub fn id_to_name(&self, id: IdResult) -> ResolvedIdName<'_> {
256        match self.id_to_name.get(&id).unwrap_or(&IdName::RawId) {
257            IdName::RawId => ResolvedIdName::Raw(id),
258            IdName::ExplicitName(name) | IdName::DerivedName(name) => ResolvedIdName::Named(name),
259        }
260    }
261}

263/// An instruction set that provides additional context information for disassembly.
264pub trait InstSetDisCtx: InstEncoding {
265    /// Add context to the supplied [`DisContext`] by modifying the various public members of it.
266    ///
267    /// Only supplies a [`RawInstSlice`] instead of a full [`InstSlice`] that has been error checked, as to allow
268    /// disassembly of partially corrupt instructions. Any [`DecodeError`] that arrises during decode should be silently
269    /// ignored. We recommend using `slice.iter().try_decode::<CoreInstSet>().skip_errors()` to decode instructions
270    /// from the [`RawInstSlice`].
271    ///
272    /// [`DecodeError`]: crate::binary::DecodeError
273    fn add_context(slice: &RawInstSlice, ctx: &mut DisContext);
274}

289/// Either [`DisContext`] or [`DisOptions`] that is turned into one via [`DisContext::no_context`]. Actual context needs
290/// to be gathered yourself, see warning in [`DisContext::no_context`].
291pub trait IntoDisContext {
292    fn into_dis_ctx(self) -> DisContext;
293}
294
295impl IntoDisContext for DisContext {
296    fn into_dis_ctx(self) -> DisContext {
297        self
298    }
299}

303        DisContext::no_context(self)
304    }

307/// A sequence of words that has been pre-processed and may be [`Display`]ed.
308///
309/// The `ISA: `[`InstEncoding`] generic determines for which instruction set these Words are disassembled.
310pub struct DisInstSlice<'a, ISA: InstSetDisCtx> {
311    slice: &'a RawInstSlice,
312    ctx: DisContext,
313    _phantom: PhantomData<ISA>,
314}
315
316impl<'a, ISA: InstSetDisCtx> DisInstSlice<'a, ISA> {
317    #[inline]
318    pub fn new(slice: impl Into<&'a RawInstSlice>, ctx: impl IntoDisContext) -> Self {
319        let slice = slice.into();
320        let mut ctx = ctx.into_dis_ctx();
321        ctx.add_context_raw::<ISA>(slice);
322        Self::new_no_context(slice, ctx)
323    }
324
325    #[inline]
326    pub fn new_no_context(slice: impl Into<&'a RawInstSlice>, ctx: impl IntoDisContext) -> Self {
327        Self {
328            slice: slice.into(),
329            ctx: ctx.into_dis_ctx(),
330            _phantom: PhantomData,
331        }
332    }
333}
334
335impl<'a, ISA: InstSetDisCtx> Display for DisInstSlice<'a, ISA> {
336    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
337        profiling::function_scope!();
338        for maybe_reader in self.slice.iter() {
339            match maybe_reader.and_then(|reader| ISA::decode(reader)) {
340                Ok(inst) => writeln!(f, "{}", inst.dis(&self.ctx))?,
341                Err(e) => writeln!(f, "Error: {}", e)?,
342            }
343        }
344        Ok(())
345    }
346}

348#[allow(clippy::match_same_arms)]
349pub fn escape_id_name(str: Cow<'_, str>) -> Option<Cow<'_, str>> {
350    profiling::function_scope!();
351    let escaped = escape_cow(str, |c| match c {
352        'A'..='Z' => c,
353        'a'..='z' => c,
354        '0'..='9' => c,
355        _ => '_',
356    });
357    if escaped.chars().all(|c| c == '_') {
358        None
359    } else {
360        Some(escaped)
361    }
362}

364/// escape chars individually with another char
365pub fn escape_cow(str: Cow<'_, str>, map: impl Fn(char) -> char) -> Cow<'_, str> {
366    profiling::function_scope!();
367    let no_escape_needed = str.chars().all(|c| map(c) == c);
368    if no_escape_needed {
369        str
370    } else {
371        // We can assume names are (mostly) ascii, so this capacity should match. If not, it'll just be one realloc.
372        let mut ret = String::with_capacity(str.len());
373        ret.extend(str.chars().map(map));
374        Cow::Owned(ret)
375    }
376}
377
378/// escape chars individually with a `&str` that may contain multiple chars
379pub fn escape_cow_str(str: Cow<'_, str>, map: impl Fn(&str) -> &str) -> Cow<'_, str> {
380    profiling::function_scope!();
381
382    let no_escape_needed = StrCharIter::new(&str).all(|c| map(c) == c);
383    if no_escape_needed {
384        str
385    } else {
386        // We can assume names are (mostly) ascii, so this capacity should match. If not, it'll just be one realloc.
387        let mut ret = String::with_capacity(str.len());
388        ret.extend(StrCharIter::new(&str).map(map));
389        Cow::Owned(ret)
390    }
391}

393/// Like a [`str::chars`] Iterator, but yields `&str` of the original `&str` instead
394pub struct StrCharIter<'a> {
395    str: &'a str,
396}
397
398impl<'a> StrCharIter<'a> {
399    pub fn new(str: &'a str) -> Self {
400        Self { str }
401    }
402}
403
404impl<'a> Iterator for StrCharIter<'a> {
405    type Item = &'a str;
406
407    fn next(&mut self) -> Option<Self::Item> {
408        let len = self.str.chars().next()?.len_utf8();
409        let (char_str, remaining) = self.str.split_at(len);
410        self.str = remaining;
411        Some(char_str)
412    }
413}

415/// Utility struct for inserting separators between variants. Use it like `format!({sep}{value})`, will skip the
416/// separator on first write.
417///
418/// Used by bitmasks to add `|` in disassembly.
419pub struct SeparatorJoiner<'a> {
420    first: Cell<bool>,
421    sep: &'a str,
422}
423
424impl<'a> SeparatorJoiner<'a> {
425    #[inline]
426    pub fn new(sep: &'a str) -> Self {
427        Self {
428            first: Cell::new(false),
429            sep,
430        }
431    }
432}
433
434impl<'a> Display for SeparatorJoiner<'a> {
435    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
436        if self.first.replace(true) {
437            write!(f, "{}", self.sep)
438        } else {
439            Ok(())
440        }
441    }
