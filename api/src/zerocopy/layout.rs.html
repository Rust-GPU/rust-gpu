16/// The target pointer width, counted in bits.
17const POINTER_WIDTH_BITS: usize = mem::size_of::<usize>() * 8;

19/// The layout of a type which might be dynamically-sized.
20///
21/// `DstLayout` describes the layout of sized types, slice types, and "slice
22/// DSTs" - ie, those that are known by the type system to have a trailing slice
23/// (as distinguished from `dyn Trait` types - such types *might* have a
24/// trailing slice type, but the type system isn't aware of it).
25///
26/// Note that `DstLayout` does not have any internal invariants, so no guarantee
27/// is made that a `DstLayout` conforms to any of Rust's requirements regarding
28/// the layout of real Rust types or instances of types.
29#[doc(hidden)]
30#[allow(missing_debug_implementations, missing_copy_implementations)]
31#[cfg_attr(any(kani, test), derive(Debug, PartialEq, Eq))]
32#[derive(Copy, Clone)]
33#[cfg_attr(kani, derive(kani::Arbitrary))]
34pub struct DstLayout {
35    pub(crate) align: NonZeroUsize,
36    pub(crate) size_info: SizeInfo,
37    // Is it guaranteed statically (without knowing a value's runtime metadata)
38    // that the top-level type contains no padding? This does *not* apply
39    // recursively - for example, `[(u8, u16)]` has `statically_shallow_unpadded
40    // = true` even though this type likely has padding inside each `(u8, u16)`.
41    pub(crate) statically_shallow_unpadded: bool,
42}

52/// The alignment and phase of a rounded size calculation.
53///
54/// If `encoded` is the stored non-zero integer, then its highest set bit is a
55/// power-of-two alignment `align`, and its remaining lower bits are a `phase`
56/// in `0..align`. Thus every non-zero bit pattern is a valid encoding, and:
57///
58/// ```text
59/// encoded = align + phase
60/// ```
61#[cfg_attr(any(kani, test), derive(Debug, PartialEq, Eq))]
62#[repr(transparent)]
63#[derive(Copy, Clone)]
64#[cfg_attr(kani, derive(kani::Arbitrary))]
65pub(crate) struct RoundingAlignAndPhase(NonZeroUsize);
66
67impl RoundingAlignAndPhase {
68    /// Encodes `align` and `phase`.
69    ///
70    /// # Panics
71    ///
72    /// Panics if `align` is not a power of two and `phase` is not less than
73    /// `align`.
74    #[inline(always)]
75    #[cfg_attr(kani, kani::requires(align.is_power_of_two() && phase < align.get()))]
76    #[cfg_attr(kani, kani::ensures(|result| result.0.get() == align.get() + phase))]
77    pub(crate) const fn new(align: NonZeroUsize, phase: usize) -> Self {
78        #[cfg(kani)]
79        #[kani::proof_for_contract(RoundingAlignAndPhase::new)]
80        #[kani::solver(kissat)]
81        fn proof() {
82            RoundingAlignAndPhase::new(kani::any(), kani::any());
83        }
84
85        const_assert!(align.get().is_power_of_two());
86        const_assert!(phase < align.get());
87
88        // Since `align` is a power of two and `phase < align`, their set bits
89        // are disjoint. The result is therefore non-zero and losslessly stores
90        // both components.
91        let encoded = align.get() | phase;
92        match NonZeroUsize::new(encoded) {
93            Some(encoded) => Self(encoded),
94            None => const_unreachable!(),
95        }
96    }
97
98    /// Decodes the alignment and phase.
99    #[inline(always)]
100    #[cfg_attr(kani, kani::ensures(|&(align, phase)| {
101        align.is_power_of_two() && phase < align.get()
102            && align.get().checked_add(phase) == Some(self.0.get())
103    }))]
104    pub(crate) const fn components(self) -> (NonZeroUsize, usize) {
105        #[cfg(kani)]
106        #[kani::proof_for_contract(RoundingAlignAndPhase::components)]
107        #[kani::solver(kissat)]
108        fn proof() {
109            kani::any::<RoundingAlignAndPhase>().components();
110        }
111        // `leading_zeros <= POINTER_WIDTH_BITS - 1` because the encoded value
112        // is non-zero. Converting `leading_zeros` to `usize` cannot truncate
113        // because it is at most the number of bits in a `usize`.
114        #[allow(
115            clippy::arithmetic_side_effects,
116            clippy::as_conversions,
117            clippy::needless_nonzero_get
118        )]
119        let shift = (POINTER_WIDTH_BITS - 1) - (self.0.get().leading_zeros() as usize);
120        #[allow(clippy::arithmetic_side_effects)]
121        let align = 1usize << shift;
122        let align = match NonZeroUsize::new(align) {
123            Some(align) => align,
124            None => const_unreachable!(),
125        };
126        // `align` is the highest set bit, so XOR removes exactly that bit and
127        // leaves a value in `0..align`.
128        let phase = self.0.get() ^ align.get();
129        (align, phase)
130    }
131
132    /// Decodes the alignment, which is the highest set bit of `self`.
133    #[inline(always)]
134    #[cfg_attr(kani, kani::ensures(|align| {
135        align.is_power_of_two() && align.get() <= self.0.get()
136            && self.0.get() - align.get() < align.get()
137    }))]
138    pub(crate) const fn align(self) -> NonZeroUsize {
139        #[cfg(kani)]
140        #[kani::proof_for_contract(RoundingAlignAndPhase::align)]
141        #[kani::solver(kissat)]
142        fn proof() {
143            kani::any::<RoundingAlignAndPhase>().align();
144        }
145
146        self.components().0
147    }
148}

153pub(crate) struct TrailingSliceLayout<E = usize> {
154    /// The offset of the first byte of the trailing slice field. Note that this
155    /// is NOT the same as the minimum size of the type. For example, consider
156    /// the following type:
157    ///
158    /// ```ignore
159    /// #[repr(C, align(2))]
160    /// struct Foo {
161    ///     header: u16,
162    ///     tag: u8,
163    ///     tail: [u8],
164    /// }
165    /// ```
166    ///
167    /// In `Foo`, `tail` is at byte offset 3. When `tail.len() == 0`, `tail` is
168    /// followed by a padding byte.
169    pub(crate) offset: usize,
170    /// The size of the element type of the trailing slice field.
171    pub(crate) elem_size: E,
172    /// The trailing slice's byte offset plus fixed trailing padding, minus the
173    /// rounding phase.
174    ///
175    /// Fixed trailing padding excludes padding added by the final alignment
176    /// rounding. The rounding alignment and phase are stored in
177    /// `size_rounding_align_and_phase`. This value is independent of the slice
178    /// length and can be smaller than, equal to, or larger than `offset`.
179    ///
180    /// For a completed layout, this value is a multiple of `DstLayout::align`.
181    /// The minimum object size equals this value when the phase is zero;
182    /// otherwise, it exceeds this value by one rounding-alignment unit.
183    pub(crate) size_base: usize,
184    /// The alignment and byte phase of the rounding applied to the trailing
185    /// slice's contribution to object size.
186    ///
187    /// The alignment `size_align` is a nonzero power of two. The phase
188    /// `size_phase` is the fixed byte contribution inside that rounding, in
189    /// `0..size_align`. Both are independent of the trailing slice's length.
190    /// For `elems` trailing elements, the enclosing object's size is:
191    ///
192    /// ```text
193    /// size_base + round_up(size_phase + elems * elem_size, size_align)
194    /// ```
195    ///
196    /// For a completed layout, `size_align` is at least `DstLayout::align`.
197    /// Packing can reduce the enclosing type's alignment and the alignment
198    /// used to place its trailing field without removing padding required by
199    /// a nested DST's own layout. For example, a `repr(packed(2))` wrapper
200    /// around a DST with alignment 4 can have an object alignment of 2 and a
201    /// rounding alignment of 4.
202    ///
203    /// The rounding parameters account for padding preserved within nested
204    /// fields. Their rounding origin can differ from the physical start of
205    /// the trailing slice, so the phase need not equal `offset % size_align`.
206    pub(crate) size_rounding_align_and_phase: RoundingAlignAndPhase,
207}

209impl<E> TrailingSliceLayout<E> {
210    /// Produces `self.size_base` rounded down to a multiple of `size_align`,
211    /// plus `size_phase`, where both components come from
212    /// `self.size_rounding_align_and_phase`.
213    ///
214    /// This value need not equal `self.offset`, the trailing slice's byte
215    /// offset.
216    #[inline(always)]
217    #[cfg_attr(kani, kani::ensures(|&offset| {
218        let (align, phase) = self.size_rounding_align_and_phase.components();
219        offset == util::round_down_to_next_multiple_of_alignment(self.size_base, align) + phase
220    }))]
221    const fn size_offset(&self) -> usize {
222        #[cfg(all(kani, kani_slow))]
223        #[kani::proof_for_contract(TrailingSliceLayout::<usize>::size_offset)]
224        #[kani::solver(kissat)]
225        fn proof() {
226            kani::any::<TrailingSliceLayout>().size_offset();
227        }
228
229        let (size_align, size_phase) = self.size_rounding_align_and_phase.components();
230        let aligned_base =
231            util::round_down_to_next_multiple_of_alignment(self.size_base, size_align);
232        // The aligned base and phase have disjoint set bits.
233        aligned_base | size_phase
234    }
235
236    /// Produces the largest trailing-slice byte capacity whose complete, padded
237    /// object fits in `available_bytes`, or `None` if the object does not fit
238    /// even with an empty trailing slice.
239    ///
240    /// This capacity is independent of `elem_size`: it need not represent a
241    /// whole number of elements, even for a valid Rust layout. For example, a
242    /// slice of `[u8; 3]` elements has three-byte elements and alignment 1.
243    /// With `available_bytes = 4`, this method produces `Some(4)`, although
244    /// only one whole element, occupying three bytes, fits.
245    #[inline(always)]
246    #[cfg_attr(kani, kani::ensures(|&result| {
247        let (align, phase) = self.size_rounding_align_and_phase.components();
248        result == available_bytes.checked_sub(self.size_base)
249            .map(|bytes| util::round_down_to_next_multiple_of_alignment(bytes, align))
250            .and_then(|rounded| rounded.checked_sub(phase))
251    }))]
252    const fn max_trailing_bytes(&self, available_bytes: usize) -> Option<usize> {
253        #[cfg(all(kani, kani_slow))]
254        #[kani::proof_for_contract(TrailingSliceLayout::<usize>::max_trailing_bytes)]
255        #[kani::solver(kissat)]
256        fn proof() {
257            kani::any::<TrailingSliceLayout>().max_trailing_bytes(kani::any());
258        }
259
260        #[cfg(all(kani, kani_slow))]
261        #[kani::proof_for_contract(TrailingSliceLayout::<NonZeroUsize>::max_trailing_bytes)]
262        #[kani::solver(kissat)]
263        fn proof_nonzero() {
264            kani::any::<TrailingSliceLayout<NonZeroUsize>>().max_trailing_bytes(kani::any());
265        }
266
267        let (size_align, size_phase) = self.size_rounding_align_and_phase.components();
268
269        // Interpret the normalized size formula over the nonnegative integers:
270        //
271        //   object_size(trailing_bytes)
272        //       = self.size_base
273        //           + round_up(size_phase + trailing_bytes, size_align).
274        //
275        // We seek the largest byte count with
276        // `object_size(trailing_bytes) <= available_bytes`.
277        //
278        // At zero trailing bytes, the input to rounding is just `size_phase`.
279        // Since `0 <= size_phase < size_align`, the least alignment multiple at
280        // or above it is zero for a zero phase, and `size_align` otherwise.
281        // Thus `rounded_phase = round_up(size_phase, size_align)` is exactly
282        // the rounded contribution of an empty tail.
283        let rounded_phase = if size_phase == 0 { 0 } else { size_align.get() };
284
285        // Adding the fixed base gives `minimum_size = object_size(0)`. The size
286        // formula is nondecreasing in the trailing byte count, so this is the
287        // minimum object size. If this sum exceeds `usize::MAX`, it also
288        // exceeds `available_bytes`, and no tail can fit.
289        let minimum_size = match self.size_base.checked_add(rounded_phase) {
290            Some(size) => size,
291            None => return None,
292        };
293
294        // Subtracting the minimum size gives the budget for increasing the
295        // rounded contribution. Failure means even the minimum size exceeds the
296        // available space. Success establishes `minimum_size + extra_bytes =
297        // available_bytes`.
298        let extra_bytes = match available_bytes.checked_sub(minimum_size) {
299            Some(bytes) => bytes,
300            None => return None,
301        };
302
303        // The rounded contribution can increase only by multiples of
304        // `size_align`. Its largest permitted increase is therefore
305        // `aligned_bytes = round_down(extra_bytes, size_align)`, with:
306        //
307        //   aligned_bytes <= extra_bytes < aligned_bytes + size_align.
308        //
309        // Hence `rounded_phase + aligned_bytes` is the largest alignment
310        // multiple that fits after the base: adding the base gives at most
311        // `minimum_size + extra_bytes = available_bytes`, while the next
312        // alignment multiple would exceed that budget.
313        let aligned_bytes = util::round_down_to_next_multiple_of_alignment(extra_bytes, size_align);
314
315        // Rounding an empty tail adds `rounded_phase - size_phase` bytes. This
316        // is `initial_padding`: the tail can use these bytes without increasing
317        // the rounded contribution. The two cases for `rounded_phase` give `0
318        // <= initial_padding < size_align`, so the subtraction is
319        // representable.
320        #[allow(clippy::arithmetic_side_effects)]
321        let initial_padding = rounded_phase - size_phase;
322
323        // For any nonnegative candidate byte count, the aligned limit above
324        // gives the following equivalences:
325        //
326        //   object_size(candidate_bytes) <= available_bytes
327        //       iff round_up(size_phase + candidate_bytes, size_align)
328        //           <= rounded_phase + aligned_bytes
329        //       iff size_phase + candidate_bytes
330        //           <= rounded_phase + aligned_bytes
331        //       iff candidate_bytes <= aligned_bytes + initial_padding.
332        //
333        // Removing `round_up` is valid because its upper bound is an alignment
334        // multiple. Thus `aligned_bytes + initial_padding` is exactly the
335        // largest tail that fits. Since `aligned_bytes` is a multiple of the
336        // power-of-two `size_align` and `initial_padding < size_align`, their
337        // set bits are disjoint. Bitwise OR therefore equals their sum, with no
338        // carry or overflow.
339        let trailing_bytes = aligned_bytes | initial_padding;
340        Some(trailing_bytes)
341    }
342}

344impl TrailingSliceLayout {
345    /// Returns the difference between the complete object size and the
346    /// trailing-slice end for `elems` elements, modulo `usize::MAX + 1`.
347    ///
348    /// If the object size fits in `usize` and contains the trailing slice,
349    /// this is the exact trailing padding. Other inputs need not describe a
350    /// Rust type; their result is still the modular difference.
351    #[inline(always)]
352    #[cfg_attr(kani, kani::ensures(|&padding| {
353        let (align, phase) = self.size_rounding_align_and_phase.components();
354        let trailing_bytes = elems.wrapping_mul(self.elem_size);
355        let mask = align.get() - 1;
356        let rounded = phase.wrapping_add(trailing_bytes).wrapping_add(mask) & !mask;
357        let object_size = self.size_base.wrapping_add(rounded);
358        padding == object_size.wrapping_sub(self.offset.wrapping_add(trailing_bytes))
359    }))]
360    pub(crate) const fn padding_for_elems(self, elems: usize) -> usize {
361        #[cfg(all(kani, kani_slow))]
362        #[kani::proof_for_contract(TrailingSliceLayout::padding_for_elems)]
363        #[kani::solver(kissat)]
364        fn proof() {
365            kani::any::<TrailingSliceLayout>().padding_for_elems(kani::any());
366        }
367
368        let (size_align, size_phase) = self.size_rounding_align_and_phase.components();
369
370        // Since `size_align` is a nonzero power of two, subtracting one gives
371        // a mask selecting precisely the remainder modulo `size_align`.
372        #[allow(clippy::arithmetic_side_effects)]
373        let size_mask = size_align.get() - 1;
374
375        // Removing whole alignment multiples from each element's size
376        // preserves the trailing-byte count modulo `size_align`. If elements
377        // occupy whole alignment units, this is zero, making the padding's
378        // independence from the element count explicit to the optimizer.
379        let elem_remainder = self.elem_size & size_mask;
380
381        // The product is congruent to `elems * self.elem_size` modulo
382        // `size_align`. Wrapping preserves that congruence because every
383        // power-of-two `size_align` divides `usize::MAX + 1`. Masking then
384        // gives the exact trailing-byte remainder in `0..size_align`.
385        let trailing_remainder = elems.wrapping_mul(elem_remainder) & size_mask;
386
387        // Adding `size_phase` gives the same remainder as the mathematical
388        // `size_phase + elems * self.elem_size`. Reducing before adding the
389        // phase also exposes this periodicity to the optimizer.
390        let rounding_input = size_phase.wrapping_add(trailing_remainder);
391
392        // `rounding_padding` is the unique value in `0..size_align` that
393        // rounds `rounding_input` up to an alignment multiple. Since only
394        // the remainder determines this value, it is also the padding added
395        // by `round_up(size_phase + elems * self.elem_size, size_align)`.
396        let rounding_padding = util::padding_needed_for(rounding_input, size_align);
397
398        // Write `trailing_bytes = elems * self.elem_size` over mathematical
399        // integers. Subtracting the trailing-slice end from the complete
400        // object size cancels `trailing_bytes`:
401        //
402        //   size_base + round_up(size_phase + trailing_bytes, size_align)
403        //       - (offset + trailing_bytes)
404        //       = size_base + size_phase - offset + rounding_padding.
405        //
406        // Wrapping computes this difference modulo `usize::MAX + 1`, even
407        // if an intermediate value or the difference is unrepresentable.
408        // If the object size fits in `usize` and contains the trailing slice,
409        // the difference is in `0..=usize::MAX`; its residue is then the
410        // exact number of padding bytes after the trailing slice.
411        self.size_base
412            .wrapping_add(size_phase)
413            .wrapping_sub(self.offset)
414            .wrapping_add(rounding_padding)
415    }
416
417    /// Computes the size described by this layout for `elems` trailing slice
418    /// elements.
419    ///
420    /// Returns `None` if the element-byte count or complete rounded size
421    /// cannot be represented as a `usize`.
422    #[inline(always)]
423    #[cfg_attr(kani, kani::ensures(|&size| {
424        size == elems.checked_mul(self.elem_size)
425            .and_then(|bytes| proofs::slice_dst_size_for_trailing_bytes(self, bytes))
426    }))]
427    pub(crate) const fn size_for_elems(self, elems: usize) -> Option<usize> {
428        #[cfg(all(kani, kani_slow))]
429        #[kani::proof_for_contract(TrailingSliceLayout::size_for_elems)]
430        #[kani::solver(kissat)]
431        fn proof() {
432            kani::any::<TrailingSliceLayout>().size_for_elems(kani::any());
433        }
434
435        // Let `size_align` and `size_phase` denote the components stored in
436        // `size_rounding_align_and_phase`. The normalized layout formula,
437        // evaluated over the nonnegative integers, is:
438        //
439        //   object_size(trailing_bytes)
440        //       = self.size_base
441        //           + round_up(size_phase + trailing_bytes, size_align)
442        //
443        // Here, `round_up` gives the least multiple of `size_align` greater
444        // than or equal to its input. Thus `object_size` is nondecreasing.
445        // By the capacity helper's contract, `max_trailing_bytes` is the
446        // largest byte count for which `object_size` fits in a `usize`.
447        // Monotonicity then gives the equivalence:
448        //
449        //   object_size(trailing_bytes) <= usize::MAX
450        //       iff trailing_bytes <= max_trailing_bytes.
451        //
452        // If the helper returns `None`, even `object_size(0)` exceeds the
453        // limit, so no element count can produce a representable size.
454        let max_trailing_bytes = match self.max_trailing_bytes(usize::MAX) {
455            Some(bytes) => bytes,
456            None => return None,
457        };
458
459        // Each of the `elems` elements contributes `self.elem_size` bytes,
460        // so substitution into the formula requires their product. Checked
461        // multiplication yields that exact product, including zero for
462        // zero-sized elements. If it overflows, the complete size cannot fit
463        // either: `object_size(trailing_bytes) >= trailing_bytes` because
464        // the base, phase, and rounding increment are all nonnegative.
465        let trailing_size = match self.elem_size.checked_mul(elems) {
466            Some(bytes) => bytes,
467            None => return None,
468        };
469
470        // The equivalence above rejects exactly the remaining cases whose
471        // complete size exceeds `usize::MAX`. After this check we have
472        // `object_size(trailing_size) <= usize::MAX`.
473        if trailing_size > max_trailing_bytes {
474            return None;
475        }
476
477        // The trailing slice starts at `self.offset` and occupies
478        // `trailing_size` bytes. Thus `trailing_end` is its end offset modulo
479        // `usize::MAX + 1`. This sum need not fit: a raw layout need not
480        // describe a Rust type, so its trailing slice need not lie within
481        // the complete object size bounded above.
482        let trailing_end = self.offset.wrapping_add(trailing_size);
483
484        // `padding_for_elems` returns `object_size(trailing_size)` minus the
485        // trailing-slice end, modulo `usize::MAX + 1`. Adding the end back
486        // therefore gives `object_size(trailing_size)` modulo the same
487        // modulus. The bound above establishes that this size fits in
488        // `usize`, so its residue is the exact size specified by the layout.
489        let size = trailing_end.wrapping_add(self.padding_for_elems(elems));
490        Some(size)
491    }
492
493    /// Returns `true` only when `self` and `other` describe the same size for
494    /// every trailing slice length.
495    ///
496    /// This recognizes sufficient conditions for equality. A `false` result
497    /// also includes equivalent size sequences that this test cannot recognize.
498    #[inline(always)]
499    #[cfg_attr(kani, kani::ensures(|&same| {
500        let (self_align, self_phase) = self.size_rounding_align_and_phase.components();
501        let (other_align, other_phase) = other.size_rounding_align_and_phase.components();
502        let initial_size = proofs::slice_dst_size_for_trailing_bytes(self, 0);
503        same == (self.elem_size == other.elem_size
504            && initial_size.is_some()
505            && initial_size == proofs::slice_dst_size_for_trailing_bytes(other, 0)
506            && ((self.elem_size % self_align.get() == 0
507                && self.elem_size % other_align.get() == 0)
508                || (self_align == other_align && self_phase == other_phase)))
509    }))]
510    pub(crate) const fn has_same_size_sequence(self, other: Self) -> bool {
511        #[cfg(all(kani, kani_slow))]
512        #[kani::proof_for_contract(TrailingSliceLayout::has_same_size_sequence)]
513        #[kani::solver(kissat)]
514        fn proof() {
515            kani::any::<TrailingSliceLayout>().has_same_size_sequence(kani::any());
516        }
517
518        // For either layout, interpret its normalized size formula over the
519        // nonnegative integers:
520        //
521        //   object_size(elems)
522        //       = size_base
523        //           + round_up(size_phase + elems * elem_size, size_align).
524        //
525        // In two cases below, we establish equality of these mathematical sizes
526        // for every `elems`. Equal sizes also have the same `usize`
527        // representability bound.
528        //
529        // Both cases require a common byte contribution at each element count.
530        // Equal element sizes give that shared term, `elems * elem_size`.
531        if self.elem_size != other.elem_size {
532            return false;
533        }
534
535        // At `elems == 0`, the element contribution vanishes. Require equal,
536        // representable initial sizes; each case below then shows that
537        // increasing `elems` adds the same amount to both initial sizes.
538        match (self.size_for_elems(0), other.size_for_elems(0)) {
539            (Some(self_size), Some(other_size)) if self_size == other_size => {}
540            _ => return false,
541        }
542
543        let (self_align, self_phase) = self.size_rounding_align_and_phase.components();
544        let (other_align, other_phase) = other.size_rounding_align_and_phase.components();
545        let self_align = self_align.get();
546        let other_align = other_align.get();
547
548        // Case 1: The element size is a multiple of both alignments.
549        //
550        // For two power-of-two alignments, the larger is divisible by the
551        // smaller. Thus `max_align` is their least common multiple, and a byte
552        // count is divisible by both alignments exactly when it is divisible by
553        // `max_align`.
554        let max_align = if self_align > other_align { self_align } else { other_align };
555
556        // If the element size is divisible by both alignments, so is `elems *
557        // elem_size`. Adding an alignment multiple commutes with rounding,
558        // giving the following identity for each layout:
559        //
560        //   round_up(size_phase + elems * elem_size, size_align)
561        //       = round_up(size_phase, size_align) + elems * elem_size.
562        //
563        // Consequently, `object_size(elems) = object_size(0) + elems *
564        // elem_size`. The initial sizes and element sizes already agree, so the
565        // complete size sequences agree. This also covers zero-sized elements,
566        // whose contribution is zero for every element count.
567        #[allow(clippy::arithmetic_side_effects)]
568        if self.elem_size % max_align == 0 {
569            return true;
570        }
571
572        // Case 2: The rounding alignments and phases match.
573        //
574        // Require a common rounding alignment before comparing phases. If the
575        // alignments differ, neither case establishes equality.
576        if self_align != other_align {
577            return false;
578        }
579
580        // If the phases also agree, both `round_up` operations have the same
581        // input and alignment at every element count. In particular, their
582        // rounded terms agree at zero. Subtracting that common term from the
583        // equal initial sizes proves `self.size_base == other.size_base`. Both
584        // the fixed and rounded contributions therefore agree for every
585        // `elems`, which establishes equality of the complete size formulas.
586        self_phase == other_phase
587    }
588
589    /// Advances the size formula by `bytes` while replacing its element size.
590    ///
591    /// Let `(size_align, size_phase)` be decoded from
592    /// `self.size_rounding_align_and_phase`. For a trailing slice length
593    /// `elems`, the returned layout describes:
594    ///
595    /// ```text
596    /// self.size_base
597    ///     + round_up(size_phase + bytes + elems * elem_size, size_align)
598    /// ```
599    ///
600    /// Here, `elem_size` is the replacement element size passed to this
601    /// method. The returned layout preserves the physical slice offset
602    /// `self.offset` and the rounding alignment; advancing changes the size
603    /// calculation.
604    ///
605    /// Returns `None` if advancing cannot produce representable normalized
606    /// components. Even when those components fit, evaluating the rounded
607    /// object size may overflow for some or all trailing slice lengths.
608    #[inline(always)]
609    #[cfg_attr(kani, kani::ensures(|result| {
610        let expected_offset = self.size_offset().checked_add(bytes);
611        match result {
612            Some(advanced) => {
613                let align = self.size_rounding_align_and_phase.align();
614                expected_offset == Some(advanced.size_offset())
615                    && advanced.offset == self.offset
616                    && advanced.elem_size == elem_size
617                    && advanced.size_rounding_align_and_phase.align() == align
618                    && (advanced.size_base & (align.get() - 1)) == (self.size_base & (align.get() - 1))
619            }
620            None => expected_offset.is_none(),
621        }
622    }))]
623    const fn advance(self, bytes: usize, elem_size: usize) -> Option<Self> {
624        #[cfg(all(kani, kani_slow))]
625        #[kani::proof_for_contract(TrailingSliceLayout::advance)]
626        #[kani::solver(kissat)]
627        fn proof() {
628            kani::any::<TrailingSliceLayout>().advance(kani::any(), kani::any());
629        }
630
631        let (size_align, size_phase) = self.size_rounding_align_and_phase.components();
632
633        // Advancing replaces the phase inside `round_up` with `size_phase +
634        // bytes`. To normalize this sum, separate it into whole alignment units
635        // and a remainder. The whole units can move outside `round_up` into the
636        // base; the remainder becomes the new phase.
637        //
638        // Since `size_align` is a nonzero power of two, `align_mask` selects
639        // exactly the bits representing a remainder modulo `size_align`.
640        // Nonzero `size_align` also makes this subtraction representable.
641        #[allow(clippy::arithmetic_side_effects)]
642        let align_mask = size_align.get() - 1;
643
644        // The base can increase by at most `usize::MAX - self.size_base`. Since
645        // only whole alignment units move into the base, the largest
646        // transferable contribution is:
647        //
648        //   whole_capacity
649        //       = round_down(usize::MAX - self.size_base, size_align).
650        //
651        // A remainder up to `align_mask` stays in the phase and consumes no
652        // base capacity. Thus the largest advanced phase that can be split into
653        // representable normalized components is `phase_capacity =
654        // whole_capacity + align_mask`. Setting the low bits of the remaining
655        // base capacity computes exactly that sum. Its two terms have disjoint
656        // set bits, so it cannot overflow. Any larger sum would transfer more
657        // whole units than the base can hold.
658        #[allow(clippy::arithmetic_side_effects)]
659        let phase_capacity = (usize::MAX - self.size_base) | align_mask;
660
661        // The existing phase already occupies `size_phase` bytes of this
662        // capacity. Subtracting it gives the greatest permitted increment:
663        //
664        //   bytes <= max_advance
665        //       iff size_phase + bytes <= phase_capacity.
666        //
667        // Since `phase_capacity >= align_mask >= size_phase`, the subtraction
668        // cannot underflow.
669        #[allow(clippy::arithmetic_side_effects)]
670        let max_advance = phase_capacity - size_phase;
671        if bytes > max_advance {
672            return None;
673        }
674
675        // This is the fixed term inside rounding after the requested
676        // advancement, before normalization. The check above proves `size_phase
677        // + bytes <= phase_capacity <= usize::MAX`, so the addition is
678        // representable.
679        #[allow(clippy::arithmetic_side_effects)]
680        let advanced_phase = size_phase + bytes;
681
682        // Taking the remainder retains precisely the part that cannot move
683        // outside rounding as whole alignment units. In particular, `0 <=
684        // normalized_phase < size_align`, as required for the new rounding
685        // parameters.
686        let normalized_phase = advanced_phase & align_mask;
687
688        // Rounding down removes precisely the remainder retained in the
689        // phase, so `whole_bytes = advanced_phase - normalized_phase`.
690        let whole_bytes =
691            util::round_down_to_next_multiple_of_alignment(advanced_phase, size_align);
692
693        // Because `advanced_phase <= phase_capacity`, we have
694        // `whole_bytes <= whole_capacity <= usize::MAX - self.size_base`.
695        // Adding this transferred contribution to the base therefore fits.
696        #[allow(clippy::arithmetic_side_effects)]
697        let size_base = self.size_base + whole_bytes;
698
699        // To verify the returned formula, let `trailing_size = elems *
700        // elem_size` and evaluate the following over the nonnegative integers.
701        // Since `whole_bytes` is a multiple of `size_align`, it can move back
702        // inside `round_up` without changing the result:
703        //
704        //   size_base + round_up(normalized_phase + trailing_size, size_align)
705        //       = self.size_base
706        //           + round_up(whole_bytes + normalized_phase + trailing_size, size_align)
707        //       = self.size_base
708        //           + round_up(size_phase + bytes + trailing_size, size_align).
709        //
710        // This is the requested advanced formula. The replacement element size
711        // supplies its new coefficient of `elems`; `offset` retains the
712        // caller's field-placement description.
713        Some(Self {
714            offset: self.offset,
715            elem_size,
716            size_base,
717            size_rounding_align_and_phase: RoundingAlignAndPhase::new(size_align, normalized_phase),
718        })
719    }

1100        repr_packed: Option<NonZeroUsize>,
1101        fields: &[DstLayout],
1102    ) -> DstLayout {

1104        // DST. Arbitrary-length composition is not established by this bound.
1105        #[cfg(all(kani, kani_slow))]
1106        #[kani::proof_for_contract(DstLayout::for_repr_c_struct)]
1107        #[kani::solver(kissat)]
1108        #[kani::unwind(4)]
1109        fn proof() {
1110            let fields: [DstLayout; 3] = kani::any();
1111            let len: usize = kani::any();
1112            kani::assume(len <= fields.len());
1113            let _ = DstLayout::for_repr_c_struct(kani::any(), kani::any(), &fields[..len]);
1114        }
1115
1116        let mut layout = DstLayout::new_zst(repr_align);
1117
1118        let mut i = 0;
1119        #[allow(clippy::arithmetic_side_effects)]
1120        while i < fields.len() {
1121            #[allow(clippy::indexing_slicing)]
1122            let field = fields[i];
1123            layout = layout.extend(field, repr_packed);
1124            i += 1;
1125        }
1126
1127        layout = layout.pad_to_align();
1128
1129        // SAFETY: `layout` accurately describes the layout of a `repr(C)`
1130        // struct with `repr_align` or `repr_packed` alignment modifications and
1131        // the given `fields`. The `layout` is constructed using a sequence of
1132        // invocations of `DstLayout::{new_zst,extend,pad_to_align}`. The
1133        // documentation of these items vows that invocations in this manner
1134        // will accurately describe a type, so long as:
1135        //
1136        //  - that type is `repr(C)`,
1137        //  - its fields are enumerated in the order they appear,
1138        //  - the presence of `repr_align` and `repr_packed` are correctly accounted for.
1139        //
1140        // We respect all three of these preconditions above.
1141        layout

1144    /// Like `Layout::extend`, this creates a layout that describes a record
1145    /// whose layout consists of `self` followed by `next` that includes the
1146    /// necessary inter-field padding, but not any trailing padding.
1147    ///
1148    /// In order to match the layout of a `#[repr(C)]` struct, this method
1149    /// should be invoked for each field in declaration order. To add trailing
1150    /// padding, call `DstLayout::pad_to_align` after extending the layout for
1151    /// all fields. If `self` corresponds to a type marked with
1152    /// `repr(packed(N))`, then `repr_packed` should be set to `Some(N)`,
1153    /// otherwise `None`.
1154    ///
1155    /// This method cannot be used to match the layout of a record with the
1156    /// default representation, as that representation is mostly unspecified.
1157    ///
1158    /// # Safety
1159    ///
1160    /// If a (potentially hypothetical) valid `repr(C)` Rust type begins with
1161    /// fields whose layout are `self`, and those fields are immediately
1162    /// followed by a field whose layout is `field`, then unsafe code may rely
1163    /// on `self.extend(field, repr_packed)` producing a layout that correctly
1164    /// encompasses those two components.
1165    ///
1166    /// We make no guarantees to the behavior of this method if these fragments
1167    /// cannot appear in a valid Rust type (e.g., the concatenation of the
1168    /// layouts would lead to a size larger than `isize::MAX`).
1169    #[doc(hidden)]
1170    #[must_use]
1171    #[inline]
1172    #[cfg_attr(kani, kani::requires(proofs::extended_layout(self, field, repr_packed).is_some()))]
1173    #[cfg_attr(kani, kani::ensures(|&result| {
1174        Some(result) == proofs::extended_layout(self, field, repr_packed)
1175    }))]
1176    pub const fn extend(self, field: DstLayout, repr_packed: Option<NonZeroUsize>) -> Self {
1177        #[cfg(kani)]
1178        #[kani::proof_for_contract(DstLayout::extend)]
1179        #[kani::solver(kissat)]
1180        fn proof() {
1181            let _ = kani::any::<DstLayout>().extend(kani::any(), kani::any());
1182        }
1183
1184        use util::{max, min, padding_needed_for};
1185
1186        // If `repr_packed` is `None`, there are no alignment constraints, and
1187        // the value can be defaulted to `THEORETICAL_MAX_ALIGN`.
1188        let max_align = match repr_packed {
1189            Some(max_align) => max_align,
1190            None => Self::THEORETICAL_MAX_ALIGN,
1191        };
1192
1193        const_assert!(max_align.get().is_power_of_two());
1194
1195        // We use Kani to prove that this method is robust to future increases
1196        // in Rust's maximum allowed alignment. However, if such a change ever
1197        // actually occurs, we'd like to be notified via assertion failures.
1198        #[cfg(not(kani))]
1199        {
1200            const_debug_assert!(self.align.get() <= DstLayout::CURRENT_MAX_ALIGN.get());
1201            const_debug_assert!(field.align.get() <= DstLayout::CURRENT_MAX_ALIGN.get());
1202            if let Some(repr_packed) = repr_packed {
1203                const_debug_assert!(repr_packed.get() <= DstLayout::CURRENT_MAX_ALIGN.get());
1204            }
1205        }
1206
1207        // The field's alignment is clamped by `repr_packed` (i.e., the
1208        // `repr(packed(N))` attribute, if any) [1].
1209        //
1210        // [1] Per https://doc.rust-lang.org/reference/type-layout.html#the-alignment-modifiers:
1211        //
1212        //   The alignments of each field, for the purpose of positioning
1213        //   fields, is the smaller of the specified alignment and the alignment
1214        //   of the field's type.
1215        let field_align = min(field.align, max_align);
1216
1217        // The struct's alignment is the maximum of its previous alignment and
1218        // `field_align`.
1219        let align = max(self.align, field_align);
1220
1221        let (interfield_padding, size_info) = match self.size_info {
1222            // If the layout is already a DST, we panic; DSTs cannot be extended
1223            // with additional fields.
1224            SizeInfo::SliceDst(..) => const_panic!("Cannot extend a DST with additional fields."),
1225
1226            SizeInfo::Sized { size: preceding_size } => {
1227                // Compute the minimum amount of inter-field padding needed to
1228                // satisfy the field's alignment, and offset of the trailing
1229                // field. [1]
1230                //
1231                // [1] Per https://doc.rust-lang.org/reference/type-layout.html#the-alignment-modifiers:
1232                //
1233                //   Inter-field padding is guaranteed to be the minimum
1234                //   required in order to satisfy each field's (possibly
1235                //   altered) alignment.
1236                let padding = padding_needed_for(preceding_size, field_align);
1237
1238                // This will not panic (and is proven to not panic, with Kani)
1239                // if the layout components can correspond to a leading layout
1240                // fragment of a valid Rust type, but may panic otherwise (e.g.,
1241                // combining or aligning the components would create a size
1242                // exceeding `isize::MAX`).
1243                let offset = match preceding_size.checked_add(padding) {
1244                    Some(offset) => offset,
1245                    None => const_panic!("Adding padding to `self`'s size overflows `usize`."),
1246                };
1247
1248                (
1249                    padding,
1250                    match field.size_info {
1251                        SizeInfo::Sized { size: field_size } => {
1252                            // If the trailing field is sized, the resulting layout
1253                            // will be sized. Its size will be the sum of the
1254                            // preceding layout, the size of the new field, and the
1255                            // size of inter-field padding between the two.
1256                            //
1257                            // This will not panic (and is proven with Kani to not
1258                            // panic) if the layout components can correspond to a
1259                            // leading layout fragment of a valid Rust type, but may
1260                            // panic otherwise (e.g., combining or aligning the
1261                            // components would create a size exceeding
1262                            // `usize::MAX`).
1263                            let size = match offset.checked_add(field_size) {
1264                                Some(size) => size,
1265                                None => const_panic!("`field` cannot be appended without the total size overflowing `usize`"),
1266                            };
1267                            SizeInfo::Sized { size }
1268                        }
1269                        SizeInfo::SliceDst(TrailingSliceLayout {
1270                            offset: trailing_offset,
1271                            elem_size,
1272                            size_base,
1273                            size_rounding_align_and_phase,
1274                        }) => {
1275                            // If the trailing field is dynamically sized, so too
1276                            // will the resulting layout. The offset of the trailing
1277                            // slice component is the sum of the offset of the
1278                            // trailing field and the trailing slice offset within
1279                            // that field.
1280                            //
1281                            // This will not panic (and is proven with Kani to not
1282                            // panic) if the layout components can correspond to a
1283                            // leading layout fragment of a valid Rust type, but may
1284                            // panic otherwise (e.g., combining or aligning the
1285                            // components would create a size exceeding
1286                            // `usize::MAX`).
1287                            let trailing_offset = match offset.checked_add(trailing_offset) {
1288                                Some(offset) => offset,
1289                                None => const_panic!("`field` cannot be appended without the total size overflowing `usize`"),
1290                            };
1291
1292                            // Before normalization, the composite's size is:
1293                            //
1294                            //   offset + size_base
1295                            //       + round_up(size_phase + elems * elem_size, size_align)
1296                            //
1297                            // `offset` does not participate in the rounding,
1298                            // so it can be added directly to the normalized
1299                            // base.
1300                            let size_base = match offset.checked_add(size_base) {
1301                                Some(base) => base,
1302                                None => const_panic!("`field` cannot be appended without the total size overflowing `usize`"),
1303                            };
1304
1305                            SizeInfo::SliceDst(TrailingSliceLayout {
1306                                offset: trailing_offset,
1307                                elem_size,
1308                                size_base,
1309                                size_rounding_align_and_phase,
1310                            })
1311                        }
1312                    },
1313                )
1314            }
1315        };
1316
1317        let statically_shallow_unpadded = self.statically_shallow_unpadded
1318            && field.statically_shallow_unpadded
1319            && interfield_padding == 0;
1320
1321        DstLayout { align, size_info, statically_shallow_unpadded }
1322    }

1324    /// Like `Layout::pad_to_align`, this routine rounds the size of this layout
1325    /// up to the nearest multiple of this type's alignment. For DST layouts,
1326    /// this updates the runtime size formula to account for trailing padding.
1327    ///
1328    /// In order to match the layout of a `#[repr(C)]` struct, this method
1329    /// should be invoked after the invocations of [`DstLayout::extend`].
1330    ///
1331    /// This method cannot be used to match the layout of a record with the
1332    /// default representation, as that representation is mostly unspecified.
1333    ///
1334    /// # Safety
1335    ///
1336    /// If a (potentially hypothetical) valid `repr(C)` type begins with fields
1337    /// whose layout are `self` followed only by zero or more bytes of trailing
1338    /// padding (not included in `self`), then unsafe code may rely on
1339    /// `self.pad_to_align()` producing a layout that correctly
1340    /// encapsulates the layout of that type.
1341    ///
1342    /// We make no guarantees to the behavior of this method if `self` cannot
1343    /// appear in a valid Rust type (e.g., because the addition of trailing
1344    /// padding would lead to a size larger than `isize::MAX`).
1345    #[doc(hidden)]
1346    #[must_use]
1347    #[inline]
1348    #[cfg_attr(kani, kani::requires(proofs::padded_layout(self).is_some()))]
1349    #[cfg_attr(kani, kani::ensures(|&result| Some(result) == proofs::padded_layout(self)))]
1350    pub const fn pad_to_align(self) -> Self {
1351        #[cfg(all(kani, kani_slow))]
1352        #[kani::proof_for_contract(DstLayout::pad_to_align)]
1353        #[kani::solver(kissat)]
1354        fn proof() {
1355            let _ = kani::any::<DstLayout>().pad_to_align();
1356        }
1357
1358        use util::padding_needed_for;
1359
1360        let (static_padding, size_info) = match self.size_info {
1361            // For sized layouts, we add the minimum amount of trailing padding
1362            // needed to satisfy alignment.
1363            SizeInfo::Sized { size: unpadded_size } => {
1364                let padding = padding_needed_for(unpadded_size, self.align);
1365                let size = match unpadded_size.checked_add(padding) {
1366                    Some(size) => size,
1367                    None => const_panic!("Adding padding caused size to overflow `usize`."),
1368                };
1369                (padding, SizeInfo::Sized { size })
1370            }
1371            // For DST layouts, trailing padding depends on the length of the
1372            // trailing DST and is computed at runtime. Normalize the composed
1373            // rounding operation back into the representation documented on
1374            // `TrailingSliceLayout`.
1375            SizeInfo::SliceDst(mut trailing) => {
1376                let (size_align, size_phase) = trailing.size_rounding_align_and_phase.components();
1377                if size_align.get() < self.align.get() {
1378                    // For a trailing-slice byte count `trailing_size`, the
1379                    // input layout describes:
1380                    //
1381                    //   trailing.size_base
1382                    //       + round_up(size_phase + trailing_size, size_align)
1383                    //
1384                    // Here, `size_align < self.align`. Both alignments are
1385                    // powers of two, so rounding the complete input size to
1386                    // `self.align` is equivalent to:
1387                    //
1388                    //   rounded_base = round_up(trailing.size_base, size_align)
1389                    //   fixed_contribution = rounded_base + size_phase
1390                    //   new_phase = fixed_contribution % self.align
1391                    //   new_base = fixed_contribution - new_phase
1392                    //   new_base + round_up(new_phase + trailing_size, self.align)
1393                    //
1394                    // `prove_dst_layout_pad_to_align_dst_outer_alignment_transition`
1395                    // checks the exact field transition and preserved
1396                    // invariants. Its `_formula` companion proves this
1397                    // identity for every representable trailing byte count.
1398                    let base_padding = padding_needed_for(trailing.size_base, size_align);
1399                    let rounded_base = match trailing.size_base.checked_add(base_padding) {
1400                        Some(base) => base,
1401                        None => const_panic!("Adding padding caused size to overflow `usize`."),
1402                    };
1403                    let fixed_contribution = match rounded_base.checked_add(size_phase) {
1404                        Some(bytes) => bytes,
1405                        None => const_panic!("Adding padding caused size to overflow `usize`."),
1406                    };
1407                    #[allow(clippy::arithmetic_side_effects)]
1408                    let phase = fixed_contribution & (self.align.get() - 1);
1409                    // Rounding down removes `phase`, the fixed contribution
1410                    // modulo `self.align`, leaving the new base.
1411                    let size_base = util::round_down_to_next_multiple_of_alignment(
1412                        fixed_contribution,
1413                        self.align,
1414                    );
1415                    trailing.size_base = size_base;
1416                    trailing.size_rounding_align_and_phase =
1417                        RoundingAlignAndPhase::new(self.align, phase);
1418                } else {
1419                    // `size_align` is a multiple of `self.align`, so
1420                    // the inner rounded size is already aligned to
1421                    // `self.align`. Only `size_base` needs to be rounded.
1422                    // `prove_dst_layout_pad_to_align_dst_inner_rounding_transition`
1423                    // checks the exact field transition and preserved
1424                    // invariants. Its `_formula` companion proves size
1425                    // equivalence for every representable trailing byte count.
1426                    let padding = padding_needed_for(trailing.size_base, self.align);
1427                    trailing.size_base = match trailing.size_base.checked_add(padding) {
1428                        Some(base) => base,
1429                        None => const_panic!("Adding padding caused size to overflow `usize`."),
1430                    };
1431                }
1432
1433                (0, SizeInfo::SliceDst(trailing))
1434            }
1435        };
1436
1437        let statically_shallow_unpadded = self.statically_shallow_unpadded && static_padding == 0;
1438
1439        DstLayout { align: self.align, size_info, statically_shallow_unpadded }
1440    }

1546    /// Given a completed type layout, produces metadata and a split index for
1547    /// a prefix or suffix cast that satisfies the type's size and alignment
1548    /// requirements.
1549    ///
1550    /// `addr` and `bytes_len` describe the address and size of the source
1551    /// memory region, and `cast_type` selects its prefix or suffix.
1552    ///
1553    /// If the cast is valid, `validate_cast_and_convert_metadata` returns
1554    /// `Ok((elems, split_at))`. For a dynamically-sized type, `elems` is the
1555    /// maximum number of trailing slice elements for which the cast is valid.
1556    /// For sized types, `elems` is meaningless and should be ignored.
1557    /// `split_at` is the index at which to split the memory region so that the
1558    /// prefix (suffix) contains the result of the cast and the remaining
1559    /// suffix (prefix) contains the leftover bytes.
1560    ///
1561    /// There are three conditions under which a cast can fail:
1562    ///
1563    /// - The smallest possible value for the type is larger than the provided
1564    ///   memory region
1565    /// - A prefix cast is requested, and `addr` does not satisfy `self`'s
1566    ///   alignment requirement
1567    /// - A suffix cast is requested, and `addr + bytes_len` does not satisfy
1568    ///   `self`'s alignment requirement (as a consequence, since all instances
1569    ///   of the type are a multiple of its alignment, no size for the type will
1570    ///   result in a starting address which is properly aligned)
1571    ///
1572    /// # Safety
1573    ///
1574    /// When `self` accurately describes a completed type layout, including
1575    /// trailing padding, and `addr + bytes_len` fits in `usize`, callers may
1576    /// rely on the following guarantees for `Ok((elems, split_at))`:
1577    ///
1578    /// - A pointer to the type (for dynamically sized types, this includes
1579    ///   `elems` as its pointer metadata) describes an object of size `size <=
1580    ///   bytes_len`
1581    /// - If this is a prefix cast:
1582    ///   - `addr` satisfies `self`'s alignment
1583    ///   - `size == split_at`
1584    /// - If this is a suffix cast:
1585    ///   - `split_at == bytes_len - size`
1586    ///   - `addr + split_at` satisfies `self`'s alignment
1587    ///
1588    /// These guarantees require the completed-layout premise; `DstLayout`
1589    /// also permits unfinished layout fragments and arbitrary field values.
1590    /// In particular, suffix alignment relies on every object size being a
1591    /// multiple of the type's alignment.
1592    ///
1593    /// Note that this method does *not* ensure that a pointer constructed from
1594    /// its return values will be a valid pointer. In particular, this method
1595    /// does not reason about `isize` overflow, which is a requirement of many
1596    /// Rust pointer APIs, and may at some point be determined to be a validity
1597    /// invariant of pointer types themselves. This should never be a problem so
1598    /// long as the arguments to this method are derived from a known-valid
1599    /// pointer (e.g., one derived from a safe Rust reference), but it is
1600    /// nonetheless the caller's responsibility to justify that pointer
1601    /// arithmetic will not overflow based on a safety argument *other than* the
1602    /// mere fact that this method returned successfully.
1603    ///
1604    /// # Panics
1605    ///
1606    /// `validate_cast_and_convert_metadata` will panic if `self` describes a
1607    /// DST whose trailing slice element is zero-sized.
1608    ///
1609    /// If `addr + bytes_len` overflows `usize`,
1610    /// `validate_cast_and_convert_metadata` may panic, or it may return
1611    /// incorrect results. No guarantees are made about when
1612    /// `validate_cast_and_convert_metadata` will panic. The caller should not
1613    /// rely on `validate_cast_and_convert_metadata` panicking in any particular
1614    /// condition, even if `debug_assertions` are enabled.
1615    #[allow(unused)]
1616    #[inline(always)]
1617    #[cfg_attr(kani, kani::requires(addr.checked_add(bytes_len).is_some()))]
1618    #[cfg_attr(kani, kani::requires(!matches!(self.size_info,
1619        SizeInfo::SliceDst(TrailingSliceLayout { elem_size: 0, .. }))))]
1620    #[cfg_attr(kani, kani::ensures(|result| {
1621        let endpoint = match cast_type {
1622            CastType::Prefix => addr,
1623            CastType::Suffix => addr + bytes_len,
1624        };
1625        if endpoint % self.align.get() != 0 {
1626            return matches!(result, Err(MetadataCastError::Alignment));
1627        }
1628        let candidate = match self.size_info {
1629            SizeInfo::Sized { size } => {
1630                if size <= bytes_len { Some((0, size)) } else { None }
1631            }
1632            SizeInfo::SliceDst(trailing) => {
1633                proofs::max_trailing_bytes(trailing, bytes_len).and_then(|bytes| {
1634                    let elems = bytes / trailing.elem_size;
1635                    trailing.size_for_elems(elems).map(|size| (elems, size))
1636                })
1637            }
1638        };

1640            (Ok((elems, split_at)), Some((expected_elems, size))) => {

1642                    CastType::Prefix => size,
1643                    CastType::Suffix => bytes_len - size,
1644                }
1645            }
1646            (Err(MetadataCastError::Size), None) => true,
1647            _ => false,
1648        }

1650    pub(crate) const fn validate_cast_and_convert_metadata(
1651        &self,
1652        addr: usize,
1653        bytes_len: usize,
1654        cast_type: CastType,
1655    ) -> Result<(usize, usize), MetadataCastError> {
1656        #[cfg(all(kani, kani_slow))]
1657        #[kani::proof_for_contract(DstLayout::validate_cast_and_convert_metadata)]
1658        #[kani::solver(kissat)]
1659        fn proof() {
1660            let cast_type = if kani::any() { CastType::Prefix } else { CastType::Suffix };
1661            let _ = kani::any::<DstLayout>().validate_cast_and_convert_metadata(
1662                kani::any(),
1663                kani::any(),
1664                cast_type,
1665            );
1666        }
1667
1668        // `debug_assert!`, but with `#[allow(clippy::arithmetic_side_effects)]`.
1669        macro_rules! __const_debug_assert {
1670            ($e:expr $(, $msg:expr)?) => {
1671                const_debug_assert!({
1672                    #[allow(clippy::arithmetic_side_effects)]
1673                    let e = $e;
1674                    e
1675                } $(, $msg)?);
1676            };
1677        }
1678
1679        // Note that, in practice, `self` is always a compile-time constant. We
1680        // do this check earlier than needed to ensure that we always panic as a
1681        // result of bugs in the program (such as calling this function on an
1682        // invalid type) instead of allowing this panic to be hidden if the cast
1683        // would have failed anyway for runtime reasons (such as a too-small
1684        // memory region).
1685        //
1686        // FIXME(#67): Once our MSRV is 1.65, use let-else:
1687        // https://blog.rust-lang.org/2022/11/03/Rust-1.65.0.html#let-else-statements
1688        let size_info = match self.size_info.try_to_nonzero_elem_size() {
1689            Some(size_info) => size_info,
1690            None => const_panic!("attempted to cast to slice type with zero-sized element"),
1691        };
1692
1693        // Precondition
1694        __const_debug_assert!(
1695            addr.checked_add(bytes_len).is_some(),
1696            "`addr` + `bytes_len` > usize::MAX"
1697        );
1698
1699        // Alignment checks go in their own block to avoid introducing variables
1700        // into the top-level scope.
1701        {
1702            // We check alignment for `addr` (for prefix casts) or `addr +
1703            // bytes_len` (for suffix casts). For a prefix cast, the correctness
1704            // of this check is trivial - `addr` is the address the object will
1705            // live at.
1706            //
1707            // Under the documented completed-layout premise, all valid sizes
1708            // for the type are multiples of its alignment. Thus, a
1709            // validly-sized instance which lives at a validly-aligned address
1710            // must also end at a validly-aligned address. Thus, if the end
1711            // address for a suffix cast (`addr + bytes_len`) is not aligned,
1712            // then no valid start address will be aligned either.
1713            let offset = match cast_type {
1714                CastType::Prefix => 0,
1715                CastType::Suffix => bytes_len,
1716            };
1717
1718            // Addition is guaranteed not to overflow because `offset <=
1719            // bytes_len`, and `addr + bytes_len <= usize::MAX` is a
1720            // precondition of this method. Modulus is guaranteed not to divide
1721            // by zero because `self.align` is non-zero.
1722            #[allow(clippy::arithmetic_side_effects)]
1723            if (addr + offset) % self.align.get() != 0 {
1724                return Err(MetadataCastError::Alignment);
1725            }
1726        }
1727
1728        let (elems, self_bytes) = match size_info {
1729            SizeInfo::Sized { size } => {
1730                if size > bytes_len {
1731                    return Err(MetadataCastError::Size);
1732                }
1733                (0, size)
1734            }
1735            SizeInfo::SliceDst(trailing) => {
1736                let max_slice_bytes = match trailing.max_trailing_bytes(bytes_len) {
1737                    Some(bytes) => bytes,
1738                    None => return Err(MetadataCastError::Size),
1739                };
1740                // Calculate the number of elements that fit in
1741                // `max_slice_bytes`. Any remaining input bytes stay outside
1742                // the selected object unless the object's own rounding
1743                // operation consumes them as padding.
1744                //
1745                // Guaranteed not to divide by zero: `elem_size` is non-zero.
1746                let (elems, _) = max_elems_for_bytes(max_slice_bytes, trailing.elem_size);
1747                // The remainder is `max_slice_bytes - elems * elem_size`.
1748                // Writing it as a remainder exposes its bound to the
1749                // optimizer: when `elem_size <= size_align`, rounding the
1750                // remainder down to `size_align` always gives zero.
1751                let size_align = trailing.size_rounding_align_and_phase.align();
1752                // `size_phase + max_slice_bytes == max_rounded_bytes`, an
1753                // alignment multiple. The selected tail removes `unused_bytes`
1754                // from this boundary; rounding back up removes only whole
1755                // alignment units. Thus its normalized rounded size is:
1756                //
1757                //   round_up(size_phase + elems * elem_size, size_align)
1758                //       = max_rounded_bytes - round_down(unused_bytes, size_align)
1759                //
1760                // `prove_slice_dst_validator_selected_size_fits` checks this
1761                // identity and its intermediate bounds, using the relations
1762                // in `prove_max_trailing_bytes_components`. The capacity is
1763                // exact by `prove_max_trailing_bytes`; the element helper's
1764                // contract establishes the largest whole-element count.
1765                #[allow(clippy::arithmetic_side_effects)]
1766                let unused_bytes = max_slice_bytes % trailing.elem_size.get();
1767                let unused_aligned_bytes =
1768                    util::round_down_to_next_multiple_of_alignment(unused_bytes, size_align);
1769                #[allow(clippy::arithmetic_side_effects)]
1770                let max_rounded_bytes = util::round_down_to_next_multiple_of_alignment(
1771                    bytes_len - trailing.size_base,
1772                    size_align,
1773                );
1774                // `unused_aligned_bytes <= unused_bytes <= max_slice_bytes
1775                // <= max_rounded_bytes`, and `size_base + max_rounded_bytes
1776                // <= bytes_len`, so neither operation can overflow.
1777                #[allow(clippy::arithmetic_side_effects)]
1778                let self_bytes = trailing.size_base + (max_rounded_bytes - unused_aligned_bytes);
1779                (elems, self_bytes)
1780            }
1781        };
1782
1783        __const_debug_assert!(self_bytes <= bytes_len);
1784
1785        let split_at = match cast_type {
1786            CastType::Prefix => self_bytes,
1787            // Guaranteed not to underflow:
1788            // - In the `Sized` branch, only returns `size` if `size <=
1789            //   bytes_len`.
1790            // - In the `SliceDst` branch, calculates `self_bytes <=
1791            //   bytes_len`.
1792            #[allow(clippy::arithmetic_side_effects)]
1793            CastType::Suffix => bytes_len - self_bytes,
1794        };
1795
1796        Ok((elems, split_at))
1797    }
1798}

1865    //

1868    //   This results in undefined behavior when `self * rhs > usize::MAX` or
1869    //   `self * rhs < usize::MIN`, i.e. when `checked_mul` would return `None`.
1870    //
1871    let scaled = unsafe { metadata.unchecked_mul(multiple) };

1874    // representable in `usize`, and `scaled == metadata * multiple`. On
1875    // toolchains with inherent unchecked arithmetic, that is exactly the
1876    // condition under which `usize::unchecked_add` does not have undefined
1877    // behavior [1]. On older supported toolchains, method resolution selects
1878    // `polyfills::NumExt::unchecked_add`, whose safety contract requires the

1880    // separate harnesses.
1881    //
1882    // [1] Per https://doc.rust-lang.org/1.93.1/std/primitive.usize.html#method.unchecked_add:
1883    //
1884    //   This results in undefined behavior when `self + rhs > usize::MAX` or
1885    //   `self + rhs < usize::MIN`, i.e. when `checked_add` would return `None`.

2011            impl<Src: ?Sized, Dst: ?Sized> CastParams<Src, Dst> {
2012                /// Given a nonzero `dst.elem_size` that exactly divides
2013                /// `src.elem_size`, produces `true` only when the following
2014                /// metadata map preserves the complete object size for every
2015                /// source element count `src_meta`:
2016                ///
2017                /// ```text
2018                /// dst_meta = dst_base + src_meta * (src.elem_size / dst.elem_size)
2019                /// ```
2020                ///
2021                /// This recognizes sufficient conditions for equality. A
2022                /// `false` result also includes equivalent size sequences
2023                /// that this test cannot recognize, or an adjustment whose
2024                /// normalized components cannot be represented in `usize`.
2025                const fn size_sequences_match(
2026                    src: TrailingSliceLayout,
2027                    dst: TrailingSliceLayout,
2028                    dst_base: usize,
2029                ) -> bool {
2030                    let base_bytes = match dst_base.checked_mul(dst.elem_size) {
2031                        Some(bytes) => bytes,
2032                        None => return false,
2033                    };
2034                    let shifted_dst = match dst.advance(base_bytes, src.elem_size) {
2035                        Some(layout) => layout,
2036                        None => return false,
2037                    };
2038                    src.has_same_size_sequence(shifted_dst)
2039                }
2040
2041                /// Given the complete layouts of `Src` and `Dst`, produces
2042                /// parameters for a cast that preserves object size.
2043                ///
2044                /// A `Some` result establishes that `Src`'s alignment is at
2045                /// least `Dst`'s and that the metadata map preserves size for
2046                /// every valid source metadata value.
2047                ///
2048                /// Supports casts between sized types of equal size, from a
2049                /// sized type to a slice DST, and between slice DSTs. A slice
2050                /// DST destination must have nonzero-sized elements; between
2051                /// slice DSTs, its element size must exactly divide the
2052                /// source's nonzero element size.
2053                ///
2054                /// Produces `None` for unsupported casts or when this method
2055                /// cannot establish a size-preserving metadata map. Rejection
2056                /// does not establish that no such map exists.
2057                const fn try_compute(
2058                    src_layout: &DstLayout,
2059                    dst_layout: &DstLayout,
2060                ) -> Option<CastParams<Src, Dst>> {
2061                    if src_layout.align.get() < dst_layout.align.get() {
2062                        return None;
2063                    }
2064
2065                    let inner = match (src_layout.size_info, dst_layout.size_info) {
2066                        (
2067                            SizeInfo::Sized { size: src_size },
2068                            SizeInfo::Sized { size: dst_size },
2069                        ) => {
2070                            if src_size != dst_size {
2071                                return None;
2072                            }
2073
2074                            // SAFETY: We checked above that `src_size ==
2075                            // dst_size`.
2076                            CastParamsInner::SizedToSized
2077                        }
2078                        (SizeInfo::Sized { size: src_size }, SizeInfo::SliceDst(_)) => {
2079                            let dst_meta = match dst_layout.metadata_for_exact_size(src_size) {
2080                                Some(meta) => meta,
2081                                None => return None,
2082                            };
2083
2084                            // SAFETY: The preceding math ensures that a `Dst`
2085                            // with `dst_meta` addresses `src_size` bytes.
2086                            CastParamsInner::SizedToUnsized { dst_meta }
2087                        }
2088                        (SizeInfo::SliceDst(src), SizeInfo::SliceDst(dst)) => {
2089                            let dst_elem_size = if let Some(e) = NonZeroUsize::new(dst.elem_size) {
2090                                e
2091                            } else {
2092                                return None;
2093                            };
2094
2095                            if src.elem_size < dst.elem_size {
2096                                return None;
2097                            }
2098
2099                            let (elem_multiple, described_src_elem_size) =
2100                                super::max_elems_for_bytes(src.elem_size, dst_elem_size);
2101                            if described_src_elem_size != src.elem_size {
2102                                return None;
2103                            }
2104
2105                            // Prefer an exact destination-element adjustment
2106                            // between `src.size_offset()` and
2107                            // `dst.size_offset()`. This preserves offset-based
2108                            // cast compatibility and metadata selection when
2109                            // several lengths have the same rounded size. It
2110                            // also avoids choosing a length whose subsequent
2111                            // padding sequence differs from the source's.
2112                            let exact_offset_delta_elems =
2113                                match src.size_offset().checked_sub(dst.size_offset()) {
2114                                    Some(delta) => {
2115                                        let (elems, described_delta) =
2116                                            super::max_elems_for_bytes(delta, dst_elem_size);
2117                                        if described_delta == delta {
2118                                            Some(elems)
2119                                        } else {
2120                                            None
2121                                        }
2122                                    }
2123                                    None => None,
2124                                };
2125
2126                            let offset_delta_elems = match exact_offset_delta_elems {
2127                                Some(elems) if Self::size_sequences_match(src, dst, elems) => elems,
2128                                _ => {
2129                                    let src_zero_size = match src.size_for_elems(0) {
2130                                        Some(size) => size,
2131                                        None => return None,
2132                                    };
2133                                    let elems =
2134                                        match dst_layout.metadata_for_exact_size(src_zero_size) {
2135                                            Some(elems) => elems,
2136                                            None => return None,
2137                                        };
2138                                    if !Self::size_sequences_match(src, dst, elems) {
2139                                        return None;
2140                                    }
2141                                    elems
2142                                }
2143                            };
2144
2145                            CastParamsInner::UnsizedToUnsized {
2146                                // SAFETY: `size_sequences_match` proves that
2147                                // this affine metadata map preserves size.
2148                                offset_delta_elems,
2149                                // SAFETY: We checked above that this is an exact
2150                                // ratio of source to destination element size.
2151                                elem_multiple,
2152                            }
2153                        }
2154                        _ => return None,
2155                    };
2156
2157                    // SAFETY: We checked above that `src.align >= dst.align`.
2158                    Some(CastParams { inner, _src: PhantomData, _dst: PhantomData })
2159                }
2160            }

2335    /// Models the size of nested `repr(C)` structs ending in a slice.
2336    ///
2337    /// `leading` lists structs from outermost to innermost. Each tuple gives
2338    /// the packing factor, minimum alignment, and prefix size before padding
2339    /// for the final field. `trailing_elem_size` and `trailing_alignment` give
2340    /// the slice element size and alignment. An empty `leading` describes the
2341    /// slice itself.
2342    ///
2343    /// The returned closure maps a slice length to the complete size, including
2344    /// padding at every nesting level. It returns `None` only on `usize`
2345    /// overflow; sizes above `isize::MAX` are permitted by this arithmetic model.
2346    /// Packing affects field placement, preserving each field's internal
2347    /// padding, even for descriptors that Rust would reject as types.
2348    ///
2349    /// # Panics
2350    ///
2351    /// Panics if any alignment or packing factor is not a power of two, a
2352    /// minimum alignment exceeds its packing factor, or the element size is
2353    /// not a multiple of the slice alignment.
2354    fn size_for_metadata_model(
2355        leading: &[(NonZeroUsize, NonZeroUsize, usize)],
2356        trailing_elem_size: usize,
2357        trailing_alignment: NonZeroUsize,
2358    ) -> impl Fn(usize) -> Option<usize> + '_ {
2359        // Rust requires power-of-two alignments and sizes divisible by their
2360        // alignment [1]; alignment modifiers also take powers of two [2].
2361        //
2362        // [1] https://doc.rust-lang.org/1.93.0/reference/type-layout.html#size-and-alignment
2363        // [2] https://doc.rust-lang.org/1.93.0/reference/type-layout.html#the-alignment-modifiers
2364        assert!(trailing_alignment.get().is_power_of_two());
2365        assert_eq!(trailing_elem_size % trailing_alignment, 0);
2366        for &(packed, align, _) in leading {
2367            assert!(packed.get().is_power_of_two());
2368            assert!(align.get().is_power_of_two());
2369            // The descriptor requires its minimum to fit under the packing
2370            // cap. It also models combinations that Rust forbids, including
2371            // simultaneous `align` and `packed` modifiers [2].
2372            assert!(align <= packed);
2373        }
2374
2375        move |elems| {
2376            // A slice shares its array section's layout [3]. Array size is
2377            // element size times length, with the element's alignment [4].
2378            //
2379            // [3] https://doc.rust-lang.org/1.93.0/reference/type-layout.html#slice-layout
2380            // [4] https://doc.rust-lang.org/1.93.0/reference/type-layout.html#array-layout
2381            let mut size = trailing_elem_size.checked_mul(elems)?;
2382            let mut alignment = trailing_alignment;
2383
2384            // Work outward with each field's complete size, including its
2385            // trailing padding: an enclosing representation "does not change
2386            // the layout of the fields themselves" [5]. Thus packing a nested
2387            // field preserves the padding already included in `size`.
2388            //
2389            // [5] https://doc.rust-lang.org/1.93.0/reference/type-layout.html#representations
2390            for &(packed, align, prefix) in leading.iter().rev() {
2391                // Packing caps the field's alignment for placement [2].
2392                let field_align = alignment.min(packed);
2393
2394                // `repr(C)` places the next field at an aligned offset [6];
2395                // packing requires the least sufficient inter-field padding
2396                // [2]. Here `prefix` is the end of the preceding fields.
2397                //
2398                // [6] https://doc.rust-lang.org/1.93.0/reference/type-layout.html#reprc-structs
2399                let offset = prefix.checked_add(util::padding_needed_for(prefix, field_align))?;
2400
2401                // `repr(C)` uses the greatest field alignment [6], and an
2402                // explicit alignment can only raise it [2]. Here `align`
2403                // summarizes the minimum from the prefix and any explicit
2404                // alignment. Both operands are <= `packed`, so their maximum
2405                // also respects the packing cap.
2406                alignment = align.max(field_align);
2407
2408                // Advance past the complete field, then round to the struct's
2409                // alignment, following `repr(C)`'s final sizing steps [6].
2410                size = offset.checked_add(size)?;
2411                size = size.checked_add(util::padding_needed_for(size, alignment))?;
2412            }
2413            // Each enclosing size is at least its field's size. Thus any
2414            // checked multiplication or addition that overflowed above would
2415            // also make the final size unrepresentable in `usize`.
2416            Some(size)
2417        }
2418    }

3454            (layout, addr, bytes_len, cast_type): (DstLayout, usize, usize, CastType),

3466                // meaningless, but in practice we set it to 0. Callers are not
3467                // allowed to rely on this, but a lot of math is nicer if
3468                // they're able to, and some callers might accidentally do that.
3469                let sized = matches!(layout.size_info, SizeInfo::Sized { .. });

3471

3473                    SizeInfo::Sized { size } => size,

3475                        elem_size,
3476                        size_base,
3477                        size_rounding_align_and_phase,
3478                        ..

3480                        let (size_align, size_phase) = size_rounding_align_and_phase.components();
3481                        let padded_size = |elems| {
3482                            let without_padding = size_phase + elems * elem_size;
3483                            size_base
3484                                + without_padding
3485                                + util::padding_needed_for(without_padding, size_align)
3486                        };
3487

3490                        // computed the largest possible value that fits in the
3491                        // given range.

3494                    }
3495                };

3498                // `validate_cast_and_convert_metadata`.
3499                assert!(resulting_size <= bytes_len, "{}", debug_str);
3500                match cast_type {
3501                    CastType::Prefix => {
3502                        assert_eq!(addr % align, 0, "{}", debug_str);
3503                        assert_eq!(resulting_size, split_at, "{}", debug_str);
3504                    }

3528                    CastType::Prefix => 0,

3958    /// Evaluates the normalized object size for a trailing slice of
3959    /// `trailing_size` bytes:
3960    ///
3961    /// ```text
3962    /// size_base + round_up(size_phase + trailing_size, size_align)

4010    /// Models appending `field` to a sized `prefix` for `DstLayout::extend`.
4011    ///
4012    /// Places the field at the first offset after the prefix that satisfies the
4013    /// field's alignment, capped by `packed` when present. The resulting
4014    /// alignment is the maximum of the prefix's alignment and this effective
4015    /// field alignment. A DST field retains its inner rounding operation, with
4016    /// its slice offset and size base shifted by the field's placement.
4017    /// Trailing padding for the enclosing layout is left to `padded_layout`.
4018    ///
4019    /// Returns `None` for a non-power-of-two alignment, an unsized prefix, or
4020    /// arithmetic overflow. The `statically_shallow_unpadded` flag is true only
4021    /// if both inputs have it set and field placement introduces no gap.
4022    pub(super) fn extended_layout(
4023        prefix: DstLayout,
4024        field: DstLayout,
4025        packed: Option<NonZeroUsize>,
4026    ) -> Option<DstLayout> {
4027        if !prefix.align.is_power_of_two()
4028            || !field.align.is_power_of_two()
4029            || packed.map_or(false, |align| !align.is_power_of_two())
4030        {
4031            return None;
4032        }
4033        let SizeInfo::Sized { size: prefix_size } = prefix.size_info else { return None };
4034        let field_align = field.align.min(packed.unwrap_or(DstLayout::THEORETICAL_MAX_ALIGN));
4035        let field_offset = round_up(prefix_size, field_align)?;
4036        let size_info = match field.size_info {
4037            SizeInfo::Sized { size } => SizeInfo::Sized { size: field_offset.checked_add(size)? },
4038            SizeInfo::SliceDst(trailing) => SizeInfo::SliceDst(TrailingSliceLayout {
4039                offset: field_offset.checked_add(trailing.offset)?,
4040                size_base: field_offset.checked_add(trailing.size_base)?,
4041                ..trailing
4042            }),
4043        };

4045            align: prefix.align.max(field_align),

4053    /// Models adding trailing padding for `DstLayout::pad_to_align`.
4054    ///
4055    /// A sized layout rounds its size up to a multiple of `layout.align`. A DST
4056    /// instead composes that rounding with its existing size expression:
4057    ///
4058    /// ```text
4059    /// round_up(size_base + round_up(size_phase + trailing_size, size_align),
4060    ///          layout.align)
4061    /// ```
4062    ///
4063    /// Here, `size_align` and `size_phase` come from the DST's
4064    /// `size_rounding_align_and_phase`. The result encodes this expression with
4065    /// one base, phase, and rounding alignment, preserving the physical slice
4066    /// offset and element size. The `statically_shallow_unpadded` flag is
4067    /// cleared only when a sized layout gains padding; DST padding is accounted
4068    /// for by the size expression.
4069    ///
4070    /// Returns `None` if `layout.align` is not a power of two or a checked
4071    /// intermediate overflows. The contract compares this expected encoding
4072    /// with the production result; the separate padding formula proofs check
4073    /// that the encoding preserves the rounded size.
4074    pub(super) fn padded_layout(layout: DstLayout) -> Option<DstLayout> {
4075        if !layout.align.is_power_of_two() {
4076            return None;
4077        }
4078        let (size_info, no_static_padding) = match layout.size_info {
4079            SizeInfo::Sized { size } => {
4080                let rounded = round_up(size, layout.align)?;
4081                (SizeInfo::Sized { size: rounded }, rounded == size)
4082            }
4083            SizeInfo::SliceDst(trailing) => {
4084                let (inner_align, phase) = trailing.size_rounding_align_and_phase.components();
4085                let (size_base, size_rounding_align_and_phase) = if inner_align >= layout.align {
4086                    (
4087                        round_up(trailing.size_base, layout.align)?,
4088                        trailing.size_rounding_align_and_phase,
4089                    )
4090                } else {
4091                    let fixed = round_up(trailing.size_base, inner_align)?.checked_add(phase)?;
4092                    let new_phase = fixed & (layout.align.get() - 1);
4093                    (fixed - new_phase, RoundingAlignAndPhase::new(layout.align, new_phase))
4094                };
4095                (
4096                    SizeInfo::SliceDst(TrailingSliceLayout {
4097                        size_base,
4098                        size_rounding_align_and_phase,
4099                        ..trailing
4100                    }),
4101                    true,
4102                )
4103            }
4104        };
4105        Some(DstLayout {
4106            size_info,
4107            statically_shallow_unpadded: layout.statically_shallow_unpadded && no_static_padding,
4108            ..layout
4109        })
4110    }

4867                        };

4870                        None => {
4871                            kani::assume(false);
4872                            loop {}
4873                        }
4874                    };
4875                    assert_eq!(
4876                        slice_dst_size_for_trailing_bytes(composite_trailing, trailing_size),
4877                        Some(expected_size)
4878                    );

4880                    // `alloc::Layout` permits sizes which are not a multiple
4881                    // of alignment, so this represents the unpadded prefix of
4882                    // the DST through the start of its trailing slice.
4883                    let field_analog =
4884                        Layout::from_size_align(field_trailing.offset, field_align.get()).unwrap();

4886                    if let Ok((actual_composite, actual_offset)) = base_analog.extend(field_analog)
4887                    {
4888                        assert_eq!(actual_offset, offset);
4889                        assert_eq!(actual_composite.size(), composite_trailing.offset);

4893                        // and `field` cannot correspond to a real Rust type
4894                        // fragment, because such a fragment would violate
4895                        // the basic invariants of a valid Rust layout. At
4896                        // the time of writing, `DstLayout` is a little more
4897                        // permissive than `Layout`, so we don't assert
4898                        // anything in this branch (e.g., unreachability).

4901                    panic!("The extension of a layout with a DST must result in a DST.")
