1//! Use [BitWriter] to write an arbitrary amount of bits into a buffer.
2use alloc::vec::Vec;
3
4/// An interface for writing an arbitrary number of bits into a buffer. Write new bits into the buffer with `write_bits`, and
5/// obtain the output using `dump`.
6#[derive(Debug)]
7pub(crate) struct BitWriter<V: AsMut<Vec<u8>>> {
8    /// The buffer that's filled with bits
9    output: V,
10    /// holds a partially filled byte which gets put in outpu when it's fill with a write_bits call
11    partial: u64,
12    bits_in_partial: usize,
13    /// The index pointing to the next unoccupied bit. Effectively just
14    /// the number of bits that have been written into the buffer so far.
15    bit_idx: usize,
16}
17
18impl BitWriter<Vec<u8>> {
19    /// Initialize a new writer.
20    pub fn new() -> Self {
21        Self {
22            output: Vec::new(),
23            partial: 0,
24            bits_in_partial: 0,
25            bit_idx: 0,
26        }
27    }
28}
29
30impl<V: AsMut<Vec<u8>>> BitWriter<V> {
31    /// Initialize a new writer.
32    pub fn from(mut output: V) -> BitWriter<V> {
33        BitWriter {
34            bit_idx: output.as_mut().len() * 8,
35            output,
36            partial: 0,
37            bits_in_partial: 0,
38        }
39    }
40
41    /// Get the current index. Can be used to reset to this index or to later change the bits at this index
42    pub fn index(&self) -> usize {
43        self.bit_idx + self.bits_in_partial
44    }
45
46    /// Reset to an index. Currently only supports resetting to a byte aligned index
47    pub fn reset_to(&mut self, index: usize) {
48        assert!(index.is_multiple_of(8));
49        self.partial = 0;
50        self.bits_in_partial = 0;
51        self.bit_idx = index;
52        self.output.as_mut().resize(index / 8, 0);
53    }
54
55    /// Change the bits at the index. `bits` contains the ǹum_bits` new bits that should be written
56    /// Instead of the current content. `bits` *MUST* only contain zeroes in the upper bits outside of the `0..num_bits` range.
57    pub fn change_bits(&mut self, idx: usize, bits: impl Into<u64>, num_bits: usize) {
58        self.change_bits_64(idx, bits.into(), num_bits);
59    }
60
61    /// Monomorphized version of `change_bits`
62    pub fn change_bits_64(&mut self, mut idx: usize, mut bits: u64, mut num_bits: usize) {
63        self.flush();
64        assert!(idx + num_bits < self.index());
65        assert!(self.index() - (idx + num_bits) > self.bits_in_partial);
66
67        // We might be changing bits unaligned to byte borders.
68        // This means the lower bits of the first byte we are touching must stay the same
69        if !idx.is_multiple_of(8) {
70            // How many (upper) bits will change in the first byte?
71            let bits_in_first_byte = 8 - (idx % 8);
72            // We don't support only changing a few bits in the middle of a byte
73            assert!(bits_in_first_byte <= num_bits);
74            // Zero out the upper bits that will be changed while keeping the lower bits intact
75            self.output.as_mut()[idx / 8] &= 0xFFu8 >> bits_in_first_byte;
76            // Shift the bits up and put them in the now zeroed out bits
77            let new_bits = (bits << (8 - bits_in_first_byte)) as u8;
78            self.output.as_mut()[idx / 8] |= new_bits;
79            // Update the state. Note that we are now definitely working byte aligned
80            num_bits -= bits_in_first_byte;
81            bits >>= bits_in_first_byte;
82            idx += bits_in_first_byte;
83        }
84
85        assert!(idx.is_multiple_of(8));
86        // We are now byte aligned, change idx to byte resolution
87        let mut idx = idx / 8;
88
89        // Update full bytes by just shifting and extracting bytes from the bits
90        while num_bits >= 8 {
91            self.output.as_mut()[idx] = bits as u8;
92            num_bits -= 8;
93            bits >>= 8;
94            idx += 1;
95        }
96
97        // Deal with leftover bits that wont fill a full byte, keeping the upper bits of the original byte intact
98        if num_bits > 0 {
99            self.output.as_mut()[idx] &= 0xFFu8 << num_bits;
100            self.output.as_mut()[idx] |= bits as u8;
101        }
102    }
103
104    /// Simply append bytes to the buffer. Only works if the buffer was already byte aligned
105    pub fn append_bytes(&mut self, data: &[u8]) {
106        if self.misaligned() != 0 {
107            panic!("Don't append bytes when writer is misaligned")
108        }
109        self.flush();
110        self.output.as_mut().extend_from_slice(data);
111        self.bit_idx += data.len() * 8;
112    }
113
114    /// Flush temporary internal buffers to the output buffer. Only works if this is currently byte aligned
115    pub fn flush(&mut self) {
116        assert!(self.bits_in_partial.is_multiple_of(8));
117        let full_bytes = self.bits_in_partial / 8;
118        self.output
119            .as_mut()
120            .extend_from_slice(&self.partial.to_le_bytes()[..full_bytes]);
121        self.partial >>= full_bytes * 8;
122        self.bits_in_partial -= full_bytes * 8;
123        self.bit_idx += full_bytes * 8;
124    }
125
126    /// Write the lower `num_bits` from `bits` into the writer. `bits` *MUST* only contain zeroes in the upper bits outside of the `0..num_bits` range.
127    pub fn write_bits(&mut self, bits: impl Into<u64>, num_bits: usize) {
128        self.write_bits_64(bits.into(), num_bits);
129    }
130
131    /// This is the special case where we need to flush the partial buffer to the output.
132    /// Marked as cold and in a separate function so the optimizer has more information.
133    #[cold]
134    fn write_bits_64_cold(&mut self, bits: u64, num_bits: usize) {
135        assert!(self.bits_in_partial + num_bits >= 64);
136        // Fill the partial buffer so it contains 64 bits
137        let bits_free_in_partial = 64 - self.bits_in_partial;
138        let part = bits << (64 - bits_free_in_partial);
139        let merged = self.partial | part;
140        // Put the 8 bytes into the output buffer
141        self.output
142            .as_mut()
143            .extend_from_slice(&merged.to_le_bytes());
144        self.bit_idx += 64;
145        self.partial = 0;
146        self.bits_in_partial = 0;
147
148        let mut num_bits = num_bits - bits_free_in_partial;
149        let mut bits = bits >> bits_free_in_partial;
150
151        // While we are at it push full bytes into the output buffer instead of polluting the partial buffer
152        while num_bits / 8 > 0 {
153            let byte = bits as u8;
154            self.output.as_mut().push(byte);
155            num_bits -= 8;
156            self.bit_idx += 8;
157            bits >>= 8;
158        }
159
160        // The last few bits belong into the partial buffer
161        assert!(num_bits < 8);
162        if num_bits > 0 {
163            let mask = (1 << num_bits) - 1;
164            self.partial = bits & mask;
165            self.bits_in_partial = num_bits;
166        }
167    }
168
169    /// Monomorphized version of `change_bits`
170    pub fn write_bits_64(&mut self, bits: u64, num_bits: usize) {
171        if num_bits == 0 {
172            return;
173        }
174
175        if bits > 0 {
176            debug_assert!(bits.ilog2() <= num_bits as u32);
177        }
178
179        // fill partial byte first
180        if num_bits + self.bits_in_partial < 64 {
181            let part = bits << self.bits_in_partial;
182            let merged = self.partial | part;
183            self.partial = merged;
184            self.bits_in_partial += num_bits;
185        } else {
186            // If the partial buffer can't hold the num_bits we need to make space
187            self.write_bits_64_cold(bits, num_bits);
188        }
189    }
190
191    /// Returns the populated buffer that you've been writing bits into.
192    ///
193    /// This function consumes the writer, so it cannot be used after
194    /// dumping
195    pub fn dump(mut self) -> V {
196        if self.misaligned() != 0 {
197            panic!("`dump` was called on a bit writer but an even number of bytes weren't written into the buffer. Was: {}", self.index())
198        }
199        self.flush();
200        debug_assert_eq!(self.partial, 0);
201        self.output
202    }
203
204    /// Returns how many bits are missing for an even byte
205    pub fn misaligned(&self) -> usize {
206        let idx = self.index();
207        if idx.is_multiple_of(8) {
208            0
209        } else {
210            8 - (idx % 8)
211        }
212    }
213}

220    #[test]
221    fn from_existing() {
222        // Define an existing vec, write some bits into it
223        let mut existing_vec = vec![255_u8];
224        let mut bw = BitWriter::from(&mut existing_vec);
225        bw.write_bits(0u8, 8);
226        bw.flush();
227        assert_eq!(vec![255, 0], existing_vec);
228    }

230    #[test]
231    fn change_bits() {
232        let mut writer = BitWriter::new();
233        writer.write_bits(0u32, 24);
234        writer.change_bits(8, 0xFFu8, 8);
235        assert_eq!(vec![0, 0xFF, 0], writer.dump());
236
237        let mut writer = BitWriter::new();
238        writer.write_bits(0u32, 24);
239        writer.change_bits(6, 0x0FFFu16, 12);
240        assert_eq!(vec![0b11000000, 0xFF, 0b00000011], writer.dump());
241    }

245        // Write the first 4 bits as 1s and the last 4 bits as 0s
246        // 1010 is used where values should never be read from.
247        let mut bw = BitWriter::new();
248        bw.write_bits(0b1111u8, 4);
249        bw.write_bits(0b0000u8, 4);
250        let output = bw.dump();
251        assert!(output.len() == 1, "Single byte written into writer returned a vec that wasn't one byte, vec was {} elements long", output.len());
