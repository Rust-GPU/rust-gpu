3/// Zstandard encodes some types of data in a way that the data must be read
4/// back to front to decode it properly. `BitReaderReversed` provides a
5/// convenient interface to do that.
6pub struct BitReaderReversed<'s> {
7    /// The index of the last read byte in the source.
8    index: usize,
9
10    /// How many bits have been consumed from `bit_container`.
11    bits_consumed: u8,
12
13    /// How many bits have been consumed past the end of the input. Will be zero until all the input
14    /// has been read.
15    extra_bits: usize,
16
17    /// The source data to read from.
18    source: &'s [u8],
19
20    /// The reader doesn't read directly from the source, it reads bits from here, and the container
21    /// is "refilled" as it's emptied.
22    bit_container: u64,
23}
24
25impl<'s> BitReaderReversed<'s> {
26    /// How many bits are left to read by the reader.
27    pub fn bits_remaining(&self) -> isize {
28        self.index as isize * 8 + (64 - self.bits_consumed as isize) - self.extra_bits as isize
29    }
30
31    pub fn new(source: &'s [u8]) -> BitReaderReversed<'s> {
32        BitReaderReversed {
33            index: source.len(),
34            bits_consumed: 64,
35            source,
36            bit_container: 0,
37            extra_bits: 0,
38        }
39    }
40
41    /// We refill the container in full bytes, shifting the still unread portion to the left, and filling the lower bits with new data
42    #[cold]
43    fn refill(&mut self) {
44        let bytes_consumed = self.bits_consumed as usize / 8;
45        if bytes_consumed == 0 {
46            return;
47        }
48
49        if self.index >= bytes_consumed {
50            // We can safely move the window contained in `bit_container` down by `bytes_consumed`
51            // If the reader wasn't byte aligned, the byte that was partially read is now in the highest order bits in the `bit_container`
52            self.index -= bytes_consumed;
53            // Some bits of the `bits_container` might have been consumed already because we read the window byte aligned
54            self.bits_consumed &= 7;
55            self.bit_container =
56                u64::from_le_bytes((&self.source[self.index..][..8]).try_into().unwrap());
57        } else if self.index > 0 {
58            // Read the last portion of source into the `bit_container`
59            if self.source.len() >= 8 {
60                self.bit_container = u64::from_le_bytes((&self.source[..8]).try_into().unwrap());
61            } else {
62                let mut value = [0; 8];
63                value[..self.source.len()].copy_from_slice(self.source);
64                self.bit_container = u64::from_le_bytes(value);
65            }
66
67            self.bits_consumed -= 8 * self.index as u8;
68            self.index = 0;
69
70            self.bit_container <<= self.bits_consumed;
71            self.extra_bits += self.bits_consumed as usize;
72            self.bits_consumed = 0;
73        } else if self.bits_consumed < 64 {
74            // Shift out already used bits and fill up with zeroes
75            self.bit_container <<= self.bits_consumed;
76            self.extra_bits += self.bits_consumed as usize;
77            self.bits_consumed = 0;
78        } else {
79            // All useful bits have already been read and more than 64 bits have been consumed, all we now do is return zeroes
80            self.extra_bits += self.bits_consumed as usize;
81            self.bits_consumed = 0;
82            self.bit_container = 0;
83        }
84
85        // Assert that at least `56 = 64 - 8` bits are available to read.
86        debug_assert!(self.bits_consumed < 8);
87    }
88
89    /// Read `n` number of bits from the source. Will read at most 56 bits.
90    /// If there are no more bits to be read from the source zero bits will be returned instead.
91    #[inline(always)]
92    pub fn get_bits(&mut self, n: u8) -> u64 {
93        if self.bits_consumed + n > 64 {
94            self.refill();
95        }
96
97        let value = self.peek_bits(n);
98        self.consume(n);
99        value
100    }
101
102    /// Get the next `n` bits from the source without consuming them.
103    /// Caller is responsible for making sure that `n` many bits have been refilled.
104    #[inline(always)]
105    pub fn peek_bits(&mut self, n: u8) -> u64 {
106        if n == 0 {
107            return 0;
108        }
109
110        let mask = (1u64 << n) - 1u64;
111        let shift_by = 64 - self.bits_consumed - n;
112        (self.bit_container >> shift_by) & mask
113    }
114
115    /// Get the next `n1` `n2` and `n3` bits from the source without consuming them.
116    /// Caller is responsible for making sure that `sum` many bits have been refilled.
117    #[inline(always)]
118    pub fn peek_bits_triple(&mut self, sum: u8, n1: u8, n2: u8, n3: u8) -> (u64, u64, u64) {
119        if sum == 0 {
120            return (0, 0, 0);
121        }
122
123        // all_three contains bits like this: |XXXX..XXX111122223333|
124        // Where XXX are already consumed bytes, 1/2/3 are bits of the respective value
125        // Lower bits are to the right
126        let all_three = self.bit_container >> (64 - self.bits_consumed - sum);
127
128        let mask1 = (1u64 << n1) - 1u64;
129        let shift_by1 = n3 + n2;
130        let val1 = (all_three >> shift_by1) & mask1;
131
132        let mask2 = (1u64 << n2) - 1u64;
133        let shift_by2 = n3;
134        let val2 = (all_three >> shift_by2) & mask2;
135
136        let mask3 = (1u64 << n3) - 1u64;
137        let val3 = all_three & mask3;
138
139        (val1, val2, val3)
140    }
141
142    /// Consume `n` bits from the source.
143    #[inline(always)]
144    pub fn consume(&mut self, n: u8) {
145        self.bits_consumed += n;
146        debug_assert!(self.bits_consumed <= 64);
147    }
148
149    /// Same as calling get_bits three times but slightly more performant
150    #[inline(always)]
151    pub fn get_bits_triple(&mut self, n1: u8, n2: u8, n3: u8) -> (u64, u64, u64) {
152        let sum = n1 + n2 + n3;
153        if sum <= 56 {
154            self.refill();
155
156            let triple = self.peek_bits_triple(sum, n1, n2, n3);
157            self.consume(sum);
158            return triple;
159        }
160
161        (self.get_bits(n1), self.get_bits(n2), self.get_bits(n3))
162    }
163}

165#[cfg(test)]
166mod test {
167
168    #[test]
169    fn it_works() {
170        let data = [0b10101010, 0b01010101];
171        let mut br = super::BitReaderReversed::new(&data);
172        assert_eq!(br.get_bits(1), 0);
173        assert_eq!(br.get_bits(1), 1);
174        assert_eq!(br.get_bits(1), 0);
175        assert_eq!(br.get_bits(4), 0b1010);
176        assert_eq!(br.get_bits(4), 0b1101);
177        assert_eq!(br.get_bits(4), 0b0101);
178        // Last 0 from source, three zeroes filled in
179        assert_eq!(br.get_bits(4), 0b0000);
180        // All zeroes filled in
181        assert_eq!(br.get_bits(4), 0b0000);
182        assert_eq!(br.bits_remaining(), -7);
183    }
