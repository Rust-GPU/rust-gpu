1//! Utilities and representations for a frame header.
2use crate::bit_io::BitWriter;
3use crate::common::MAGIC_NUM;
4use crate::encoding::util::{find_min_size, minify_val};
5use alloc::vec::Vec;
6
7/// A header for a single Zstandard frame.
8///
9/// <https://github.com/facebook/zstd/blob/dev/doc/zstd_compression_format.md#frame_header>
10#[derive(Debug)]
11pub struct FrameHeader {
12    /// Optionally, the original (uncompressed) size of the data within the frame in bytes.
13    /// If not present, `window_size` must be set.
14    pub frame_content_size: Option<u64>,
15    /// If set to true, data must be regenerated within a single
16    /// continuous memory segment.
17    pub single_segment: bool,
18    /// If set to true, a 32 bit content checksum will be present
19    /// at the end of the frame.
20    pub content_checksum: bool,
21    /// If a dictionary ID is provided, the ID of that dictionary.
22    pub dictionary_id: Option<u64>,
23    /// The minimum memory buffer required to compress a frame. If not present,
24    /// `single_segment` will be set to true. If present, this value must be greater than 1KB
25    /// and less than 3.75TB. Encoders should not generate a frame that requires a window size larger than
26    /// 8mb.
27    pub window_size: Option<u64>,
28}
29
30impl FrameHeader {
31    /// Writes the serialized frame header into the provided buffer.
32    ///
33    /// The returned header *does include* a frame header descriptor.
34    pub fn serialize(self, output: &mut Vec<u8>) {
35        vprintln!("Serializing frame with header: {self:?}");
36        // https://github.com/facebook/zstd/blob/dev/doc/zstd_compression_format.md#frame_header
37        // Magic Number:
38        output.extend_from_slice(&MAGIC_NUM.to_le_bytes());
39
40        // `Frame_Header_Descriptor`:
41        output.push(self.descriptor());
42
43        // `Window_Descriptor
44        // TODO: https://github.com/facebook/zstd/blob/dev/doc/zstd_compression_format.md#window_descriptor
45        if !self.single_segment {
46            if let Some(window_size) = self.window_size {
47                let log = window_size.next_power_of_two().ilog2();
48                let exponent = if log > 10 { log - 10 } else { 1 } as u8;
49                output.push(exponent << 3);
50            }
51        }
52
53        if let Some(id) = self.dictionary_id {
54            output.extend(minify_val(id));
55        }
56
57        if let Some(frame_content_size) = self.frame_content_size {
58            output.extend(minify_val_fcs(frame_content_size));
59        }
60    }
61
62    /// Generate a serialized frame header descriptor for the frame header.
63    ///
64    /// https://github.com/facebook/zstd/blob/dev/doc/zstd_compression_format.md#frame_header_descriptor
65    fn descriptor(&self) -> u8 {
66        let mut bw = BitWriter::new();
67        // A frame header starts with a frame header descriptor.
68        // It describes what other fields are present
69        // https://github.com/facebook/zstd/blob/dev/doc/zstd_compression_format.md#frame_header_descriptor
70        // Writing the frame header descriptor:
71        // `Frame_Content_Size_flag`:
72        // The Frame_Content_Size_flag specifies if
73        // the Frame_Content_Size field is provided within the header.
74        // TODO: The Frame_Content_Size field isn't set at all, we should prefer to include it always.
75        // If the `Single_Segment_flag` is set and this value is zero,
76        // the size of the FCS field is 1 byte.
77        // Otherwise, the FCS field is omitted.
78        // | Value | Size of field (Bytes)
79        // | 0     | 0 or 1
80        // | 1     | 2
81        // | 2     | 4
82        // | 3     | 8
83
84        // `Dictionary_ID_flag`:
85        if let Some(id) = self.dictionary_id {
86            let flag_value: u8 = match find_min_size(id) {
87                0 => 0,
88                1 => 1,
89                2 => 2,
90                4 => 3,
91                _ => panic!(),
92            };
93            bw.write_bits(flag_value, 2);
94        } else {
95            // A `Dictionary_ID` was not provided
96            bw.write_bits(0u8, 2);
97        }
98
99        // `Content_Checksum_flag`:
100        if self.content_checksum {
101            bw.write_bits(1u8, 1);
102        } else {
103            bw.write_bits(0u8, 1);
104        }
105
106        // `Reserved_bit`:
107        // This value must be zero
108        bw.write_bits(0u8, 1);
109
110        // `Unused_bit`:
111        // An encoder compliant with this spec must set this bit to zero
112        bw.write_bits(0u8, 1);
113
114        // `Single_Segment_flag`:
115        // If this flag is set, data must be regenerated within a single continuous memory segment,
116        // and the `Frame_Content_Size` field must be present in the header.
117        // If this flag is not set, the `Window_Descriptor` field must be present in the frame header.
118        if self.single_segment {
119            assert!(self.frame_content_size.is_some(), "if the `single_segment` flag is set to true, then a frame content size must be provided");
120            bw.write_bits(1u8, 1);
121        } else {
122            assert!(
123                self.window_size.is_some(),
124                "if the `single_segment` flag is set to false, then a window size must be provided"
125            );
126            bw.write_bits(0u8, 1);
127        }
128
129        if let Some(frame_content_size) = self.frame_content_size {
130            let field_size = find_min_size(frame_content_size);
131            let flag_value: u8 = match field_size {
132                1 => 0,
133                2 => 1,
134                4 => 2,
135                3 => 8,
136                _ => panic!(),
137            };
138
139            bw.write_bits(flag_value, 2);
140        } else {
141            // `Frame_Content_Size` was not provided
142            bw.write_bits(0u8, 2);
143        }
144
145        bw.dump()[0]
146    }
147}

149/// Identical to [`minify_val`], but it implements the following edge case:
150///
151/// > When FCS_Field_Size is 1, 4 or 8 bytes, the value is read directly. When FCS_Field_Size is 2, the offset of 256 is added.
152///
153/// https://github.com/facebook/zstd/blob/dev/doc/zstd_compression_format.md#frame_content_size
154fn minify_val_fcs(val: u64) -> Vec<u8> {
155    let new_size = find_min_size(val);
156    let mut val = val;
157    if new_size == 2 {
158        val -= 256;
159    }
160    val.to_le_bytes()[0..new_size].to_vec()
161}

169    #[test]
170    fn frame_header_descriptor_decode() {
171        let header = FrameHeader {
172            frame_content_size: Some(1),
173            single_segment: true,
174            content_checksum: false,
175            dictionary_id: None,
176            window_size: None,
177        };
178        let descriptor = header.descriptor();
179        let decoded_descriptor = FrameDescriptor(descriptor);
180        assert_eq!(decoded_descriptor.frame_content_size_bytes().unwrap(), 1);
181        assert!(!decoded_descriptor.content_checksum_flag());
182        assert_eq!(decoded_descriptor.dictionary_id_bytes().unwrap(), 0);
183    }
184
185    #[test]
186    fn frame_header_decode() {
187        let header = FrameHeader {
188            frame_content_size: Some(1),
189            single_segment: true,
190            content_checksum: false,
191            dictionary_id: None,
192            window_size: None,
193        };
194
195        let mut serialized_header = Vec::new();
196        header.serialize(&mut serialized_header);
197        let parsed_header = read_frame_header(serialized_header.as_slice()).unwrap().0;
198        assert!(parsed_header.dictionary_id().is_none());
199        assert_eq!(parsed_header.frame_content_size(), 1);
200    }
