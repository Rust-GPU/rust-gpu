19    /// Encodes the data using the provided table
20    /// Writes
21    /// * Table description
22    /// * Encoded data
23    /// * Last state index
24    /// * Padding bits to fill up last byte
25    #[cfg(any(test, feature = "fuzz_exports"))]
26    pub fn encode(&mut self, data: &[u8]) {
27        self.write_table();
28
29        let mut state = self.table.start_state(data[data.len() - 1]);
30        for x in data[0..data.len() - 1].iter().rev().copied() {
31            let next = self.table.next_state(x, state.index);
32            let diff = state.index - next.baseline;
33            self.writer.write_bits(diff as u64, next.num_bits as usize);
34            state = next;
35        }
36        self.writer
37            .write_bits(state.index as u64, self.acc_log() as usize);
38
39        let bits_to_fill = self.writer.misaligned();
40        if bits_to_fill == 0 {
41            self.writer.write_bits(1u32, 8);
42        } else {
43            self.writer.write_bits(1u32, bits_to_fill);
44        }
45    }
46
47    /// Encodes the data using the provided table but with two interleaved streams
48    /// Writes
49    /// * Table description
50    /// * Encoded data with two interleaved states
51    /// * Both Last state indexes
52    /// * Padding bits to fill up last byte
53    pub fn encode_interleaved(&mut self, data: &[u8]) {
54        self.write_table();
55
56        let mut state_1 = self.table.start_state(data[data.len() - 1]);
57        let mut state_2 = self.table.start_state(data[data.len() - 2]);
58
59        // The first two symbols are represented by the start states
60        // Then encode the state transitions for two symbols at a time
61        let mut idx = data.len() - 4;
62        loop {
63            {
64                let state = state_1;
65                let x = data[idx + 1];
66                let next = self.table.next_state(x, state.index);
67                let diff = state.index - next.baseline;
68                self.writer.write_bits(diff as u64, next.num_bits as usize);
69                state_1 = next;
70            }
71            {
72                let state = state_2;
73                let x = data[idx];
74                let next = self.table.next_state(x, state.index);
75                let diff = state.index - next.baseline;
76                self.writer.write_bits(diff as u64, next.num_bits as usize);
77                state_2 = next;
78            }
79
80            if idx < 2 {
81                break;
82            }
83            idx -= 2;
84        }
85
86        // Determine if we have an even or odd number of symbols to encode
87        // If odd we need to encode the last states transition and encode the final states in the flipped order
88        if idx == 1 {
89            let state = state_1;
90            let x = data[0];
91            let next = self.table.next_state(x, state.index);
92            let diff = state.index - next.baseline;
93            self.writer.write_bits(diff as u64, next.num_bits as usize);
94            state_1 = next;
95
96            self.writer
97                .write_bits(state_2.index as u64, self.acc_log() as usize);
98            self.writer
99                .write_bits(state_1.index as u64, self.acc_log() as usize);
100        } else {
101            self.writer
102                .write_bits(state_1.index as u64, self.acc_log() as usize);
103            self.writer
104                .write_bits(state_2.index as u64, self.acc_log() as usize);
105        }
106
107        let bits_to_fill = self.writer.misaligned();
108        if bits_to_fill == 0 {
109            self.writer.write_bits(1u32, 8);
110        } else {
111            self.writer.write_bits(1u32, bits_to_fill);
112        }
113    }
114
115    fn write_table(&mut self) {
116        self.table.write_table(self.writer);
117    }
118
119    pub(super) fn acc_log(&self) -> u8 {
120        self.table.acc_log()
121    }
122}

124#[derive(Debug, Clone)]
125pub struct FSETable {
126    /// Indexed by symbol
127    pub(super) states: [SymbolStates; 256],
128    /// Sum of all states.states.len()
129    pub(crate) table_size: usize,
130}
131
132impl FSETable {
133    pub(crate) fn next_state(&self, symbol: u8, idx: usize) -> &State {
134        let states = &self.states[symbol as usize];
135        states.get(idx, self.table_size)
136    }
137
138    pub(crate) fn start_state(&self, symbol: u8) -> &State {
139        let states = &self.states[symbol as usize];
140        &states.states[0]
141    }
142
143    pub fn acc_log(&self) -> u8 {
144        self.table_size.ilog2() as u8
145    }
146
147    pub(crate) fn write_table<V: AsMut<Vec<u8>>>(&self, writer: &mut BitWriter<V>) {
148        writer.write_bits(self.acc_log() - 5, 4);
149        let mut probability_counter = 0usize;
150        let probability_sum = 1 << self.acc_log();
151
152        let mut prob_idx = 0;
153        while probability_counter < probability_sum {
154            let max_remaining_value = probability_sum - probability_counter + 1;
155            let bits_to_write = max_remaining_value.ilog2() + 1;
156            let low_threshold = ((1 << bits_to_write) - 1) - (max_remaining_value);
157            let mask = (1 << (bits_to_write - 1)) - 1;
158
159            let prob = self.states[prob_idx].probability;
160            prob_idx += 1;
161            let value = (prob + 1) as u32;
162            if value < low_threshold as u32 {
163                writer.write_bits(value, bits_to_write as usize - 1);
164            } else if value > mask {
165                writer.write_bits(value + low_threshold as u32, bits_to_write as usize);
166            } else {
167                writer.write_bits(value, bits_to_write as usize);
168            }
169
170            if prob == -1 {
171                probability_counter += 1;
172            } else if prob > 0 {
173                probability_counter += prob as usize;
174            } else {
175                let mut zeros = 0u8;
176                while self.states[prob_idx].probability == 0 {
177                    zeros += 1;
178                    prob_idx += 1;
179                    if zeros == 3 {
180                        writer.write_bits(3u8, 2);
181                        zeros = 0;
182                    }
183                }
184                writer.write_bits(zeros, 2);
185            }
186        }
187        writer.write_bits(0u8, writer.misaligned());
188    }
189}

200        let start_search_at = (idx * self.states.len()) / max_idx;
201        self.states[start_search_at..]

203            .find(|state| state.contains(idx))
204            .unwrap()
205    }

226pub fn build_table_from_data(
227    data: impl Iterator<Item = u8>,
228    max_log: u8,
229    avoid_0_numbit: bool,
230) -> FSETable {
231    let mut counts = [0; 256];
232    let mut max_symbol = 0;
233    for x in data {
234        counts[x as usize] += 1;
235    }
236    for (idx, count) in counts.iter().copied().enumerate() {
237        if count > 0 {
238            max_symbol = idx;
239        }
240    }
241    build_table_from_counts(&counts[..=max_symbol], max_log, avoid_0_numbit)
242}

244fn build_table_from_counts(counts: &[usize], max_log: u8, avoid_0_numbit: bool) -> FSETable {
245    let mut probs = [0; 256];
246    let probs = &mut probs[..counts.len()];
247    let mut min_count = 0;
248    for (idx, count) in counts.iter().copied().enumerate() {
249        probs[idx] = count as i32;
250        if count > 0 && (count < min_count || min_count == 0) {
251            min_count = count;
252        }
253    }
254
255    // shift all probabilities down so that the lowest are 1
256    min_count -= 1;
257    let mut max_prob = 0i32;
258    for prob in probs.iter_mut() {
259        if *prob > 0 {
260            *prob -= min_count as i32;
261        }
262        max_prob = max_prob.max(*prob);
263    }
264
265    if max_prob > 0 && max_prob as usize > probs.len() {
266        let divisor = max_prob / (probs.len() as i32);
267        for prob in probs.iter_mut() {
268            if *prob > 0 {
269                *prob = (*prob / divisor).max(1)
270            }
271        }
272    }
273
274    // normalize probabilities to a 2^x
275    let sum = probs.iter().sum::<i32>();
276    assert!(sum > 0);
277    let sum = sum as usize;
278    let acc_log = (sum.ilog2() as u8 + 1).max(5);
279    let acc_log = u8::min(acc_log, max_log);
280
281    if sum < 1 << acc_log {
282        // just raise the maximum probability as much as possible
283        // TODO is this optimal?
284        let diff = (1 << acc_log) - sum;
285        let max = probs.iter_mut().max().unwrap();
286        *max += diff as i32;
287    } else {
288        // decrease the smallest ones to 1 first
289        let mut diff = sum - (1 << acc_log);
290        while diff > 0 {
291            let min = probs.iter_mut().filter(|prob| **prob > 1).min().unwrap();
292            let decrease = usize::min(*min as usize - 1, diff);
293            diff -= decrease;
294            *min -= decrease as i32;
295        }
296    }
297    let max = probs.iter_mut().max().unwrap();
298    if avoid_0_numbit && *max > 1 << (acc_log - 1) {
299        let redistribute = *max - (1 << (acc_log - 1));
300        *max -= redistribute;
301        let max = *max;
302
303        // find first occurence of the second_max to avoid lifting the last zero
304        let second_max = *probs.iter_mut().filter(|x| **x != max).max().unwrap();
305        let second_max = probs.iter_mut().find(|x| **x == second_max).unwrap();
306        *second_max += redistribute;
307        assert!(*second_max <= max);
308    }
309
310    build_table_from_probabilities(probs, acc_log)
311}

313pub(super) fn build_table_from_probabilities(probs: &[i32], acc_log: u8) -> FSETable {
314    let mut states = core::array::from_fn::<SymbolStates, 256, _>(|_| SymbolStates {
315        states: Vec::new(),
316        probability: 0,
317    });
318
319    // distribute -1 symbols
320    let mut negative_idx = (1 << acc_log) - 1;
321    for (symbol, _prob) in probs
322        .iter()
323        .copied()
324        .enumerate()
325        .filter(|prob| prob.1 == -1)
326    {
327        states[symbol].states.push(State {
328            num_bits: acc_log,
329            baseline: 0,
330            last_index: (1 << acc_log) - 1,
331            index: negative_idx,
332        });
333        states[symbol].probability = -1;
334        negative_idx -= 1;
335    }
336
337    // distribute other symbols
338
339    // Setup all needed states per symbol with their respective index
340    let mut idx = 0;
341    for (symbol, prob) in probs.iter().copied().enumerate() {
342        if prob <= 0 {
343            continue;
344        }
345        states[symbol].probability = prob;
346        let states = &mut states[symbol].states;
347        for _ in 0..prob {
348            states.push(State {
349                num_bits: 0,
350                baseline: 0,
351                last_index: 0,
352                index: idx,
353            });
354
355            idx = next_position(idx, 1 << acc_log);
356            while idx > negative_idx {
357                idx = next_position(idx, 1 << acc_log);
358            }
359        }
360        assert_eq!(states.len(), prob as usize);
361    }
362
363    // After all states know their index we can determine the numbits and baselines
364    for (symbol, prob) in probs.iter().copied().enumerate() {
365        if prob <= 0 {
366            continue;
367        }
368        let prob = prob as u32;
369        let state = &mut states[symbol];
370
371        // We process the states in their order in the table
372        state.states.sort_by_key(|l| l.index);
373
374        let prob_log = if prob.is_power_of_two() {
375            prob.ilog2()
376        } else {
377            prob.ilog2() + 1
378        };
379        let rounded_up = 1u32 << prob_log;
380
381        // The lower states target double the amount of indexes -> numbits + 1
382        let double_states = rounded_up - prob;
383        let single_states = prob - double_states;
384        let num_bits = acc_log - prob_log as u8;
385        let mut baseline = (single_states as usize * (1 << (num_bits))) % (1 << acc_log);
386        for (idx, state) in state.states.iter_mut().enumerate() {
387            if (idx as u32) < double_states {
388                let num_bits = num_bits + 1;
389                state.baseline = baseline;
390                state.num_bits = num_bits;
391                state.last_index = baseline + ((1 << num_bits) - 1);
392
393                baseline += 1 << num_bits;
394                baseline %= 1 << acc_log;
395            } else {
396                state.baseline = baseline;
397                state.num_bits = num_bits;
398                state.last_index = baseline + ((1 << num_bits) - 1);
399                baseline += 1 << num_bits;
400            }
401        }
402
403        // For encoding we use the states ordered by the indexes they target
404        state.states.sort_by_key(|l| l.baseline);
405    }
406
407    FSETable {
408        table_size: 1 << acc_log,
409        states,
410    }
411}

413/// Calculate the position of the next entry of the table given the current
414/// position and size of the table.
415fn next_position(mut p: usize, table_size: usize) -> usize {
416    p += (table_size >> 1) + (table_size >> 3) + 3;
417    p &= table_size - 1;
418    p
419}
