3//! Model transform constructors: orient objects in world space.
4//!
5//! These functions produce local -> world transforms (object rotation and translation).
6//!
7//! Unlike view transforms, model transforms take explicit local axis parameters
8//! (`fwd_axis` and `up_axis`) because object conventions vary:
9//!
10//! | Engine / Convention | `fwd_axis`    | `up_axis` |
11//! |---------------------|---------------|-----------|
12//! | Bevy / OpenGL       | `Vec3::NEG_Z` | `Vec3::Y` |
13//! | Unreal Engine       | `Vec3::X`     | `Vec3::Z` |
14//! | Unity / glTF        | `Vec3::Z`     | `Vec3::Y` |
15//! | Blender             | `Vec3::Y`     | `Vec3::Z` |
16//!
17//! `fwd_axis` and `up_axis` must be perpendicular.
18//!
19//! `look_to_*` functions target a forward direction (`dir`), `look_at_*` functions
20//! target a focal point (`center`). Functions returning `Mat4`, `Affine3`, `Affine3A`,
21//! or similar return a full model transform (rotation and translation). Functions returning
22//! `Quat`, `Mat3`, `Mat3A`, or similar return only the model rotation.
23//!
24//! # Example
25//!
26//! ```
27//! use glam::Vec3;
28//! use glam::camera::model;
29//!
30//! // Bevy: object faces -Z, up is Y
31//! let q = model::look_to_quat(Vec3::X, Vec3::Y, Vec3::NEG_Z, Vec3::Y);
32//! assert!((q * Vec3::NEG_Z - Vec3::X).length() < 1e-6);
33//! ```

37/// Removes the component of `v` parallel to the unit vector `n` and normalizes the
38/// result.
39#[inline]
40#[must_use]
41fn orthogonalize(v: Vec3, n: Vec3) -> Vec3 {
42    (v - n * v.dot(n)).normalize()
43}

45/// Returns the rotation that maps the local frame given by `fwd_axis` and `up_axis`
46/// onto the world frame given by `dir` and `up`.
47///
48/// The third local and world basis vectors are derived from the given pairs, so the
49/// handedness of the local frame is encoded in `fwd_axis` and `up_axis`.
50#[inline]
51#[must_use]
52#[cfg_attr(any(debug_assertions, feature = "glam-assert"), track_caller)]
53fn look_to_rotation(dir: Vec3, up: Vec3, fwd_axis: Vec3, up_axis: Vec3) -> Mat3 {
54    glam_assert!(dir.is_normalized());
55    glam_assert!(up.is_normalized());
56    glam_assert!(fwd_axis.is_normalized());
57    glam_assert!(up_axis.is_normalized());
58    glam_assert!(
59        fwd_axis.dot(up_axis).abs() <= 2e-4,
60        "fwd_axis and up_axis must be perpendicular"
61    );
62
63    let u = orthogonalize(up, dir);
64    let z_local = fwd_axis.cross(up_axis);
65    let z_world = dir.cross(u);
66
67    let local = Mat3::from_cols(fwd_axis, up_axis, z_local);
68    let world = Mat3::from_cols(dir, u, z_world);
69
70    world * local.transpose()
71}

73/// Returns a `Quat` model rotation from eye, focal point, and up.
74///
75/// `fwd_axis` and `up_axis` are the object's local forward and up axes. The rotation
76/// maps `fwd_axis` onto the direction from `eye` to `center` and `up_axis` onto `up`.
77///
78/// # Panics
79///
80/// Will panic if `up`, `fwd_axis` or `up_axis` are not normalized, if `fwd_axis` and
81/// `up_axis` are not perpendicular, or if `center` is
82/// equal to `eye`, or if the direction from `eye` to `center` is parallel to `up`, when
83/// `glam_assert` is enabled.
84#[inline]
85#[must_use]
86#[cfg_attr(any(debug_assertions, feature = "glam-assert"), track_caller)]
87pub fn look_at_quat(eye: Vec3, center: Vec3, up: Vec3, fwd_axis: Vec3, up_axis: Vec3) -> Quat {
88    look_to_quat((center - eye).normalize(), up, fwd_axis, up_axis)
89}
