29    pub fn update(&self, state: u32, buf: &[u8]) -> u32 {
30        // SAFETY: Calling a `#[target_feature]` function is only unsafe because the CPU might lack
31        // that feature. A `State` only exists if `new` found the `crc` feature, and
32        // `calculate_pmull` (which also needs `aes`, for `PMULL`) is only called once `has_pmull`
33        // found that too.
34        //
35        // Checking `aes` here rather than storing it from `new` keeps `State` zero-sized, which
36        // makes `hash()` of 8 to 48 bytes, which builds a `Hasher` each call, 18-35% faster on an
37        // Apple M4 (and `#[inline]` here undoes it). The short inputs go to the `crc32` chains
38        // from here, too: a second call through `calculate_pmull`, which can't be inlined into
39        // this function, costs them 20-30%.
40        unsafe {
41            if buf.len() >= MIN_PMULL_BYTES && has_pmull() {
42                calculate_pmull(state, buf)
43            } else {
44                calculate(state, buf)
45            }
46        }
47    }
48}

63// Bytes per stream in the 3-way loops. Merging the streams appends zero bytes to the earlier CRC
64// states, which for a fixed length is a multiplication by a constant: four lookups in a table built
65// at compile time. Shorter blocks would help cores with a slow `crc32x` (Apple M1), but cost a
66// little on cores where one chain already keeps up (Neoverse N2), so the 3-way loop starts at
67// 3 * SHORT = 1.5 KiB, as before.
68const LONG: usize = 32768;
69const SHORT: usize = 512;
70
71// The 3-way loop advances each chain four words (32 bytes) at a time, so blocks leave no remainder.
72const _: () = assert!(LONG % 32 == 0 && SHORT % 32 == 0);
73
74static LONG_SHIFT: [[u32; 256]; 4] = crate::combine::shift_table(LONG as u64);
75static SHORT_SHIFT: [[u32; 256]; 4] = crate::combine::shift_table(SHORT as u64);

77/// Reads the `i`-th 8-byte word of `b` as a little-endian `u64`, the byte order `crc32x` expects.
78#[inline(always)]
79fn quad(b: &[u8], i: usize) -> u64 {
80    u64::from_le_bytes(b[i * 8..i * 8 + 8].try_into().unwrap())
81}

83// target_feature is necessary to allow rustc to inline the crc32* wrappers
84#[target_feature(enable = "crc")]
85unsafe fn calculate(crc: u32, data: &[u8]) -> u32 {
86    // Split the input into unaligned leading bytes, 8-byte-aligned words, and trailing bytes.
87    let head = data.as_ptr().align_offset(8).min(data.len());
88    let (pre_quad, rest) = data.split_at(head);
89    let (mut quads, post_quad) = rest.split_at(rest.len() / 8 * 8);
90
91    // Unaligned leading bytes, then switch to the internal (bit-inverted) state.
92    let mut c0 = !crc_bytes(crc, pre_quad);
93
94    // A single `crc32d` chain is latency-bound: each instruction depends on the previous result.
95    // While enough words remain, advance three independent chains over three consecutive blocks of
96    // `n` bytes. B and C start from a zero state, so appending them to A is linear: shift A's state
97    // past B's bytes and XOR in B's state, then the same for C.
98    for &(n, shift) in &[(LONG, &LONG_SHIFT), (SHORT, &SHORT_SHIFT)] {
99        while quads.len() >= 3 * n {
100            let (a, rest) = quads.split_at(n);
101            let (b, rest) = rest.split_at(n);
102            let (c, rest) = rest.split_at(n);
103            quads = rest;
104
105            let mut c1 = 0;
106            let mut c2 = 0;
107            for ((qa, qb), qc) in a
108                .chunks_exact(32)
109                .zip(b.chunks_exact(32))
110                .zip(c.chunks_exact(32))
111            {
112                for k in 0..4 {
113                    c0 = arch::__crc32d(c0, quad(qa, k));
114                    c1 = arch::__crc32d(c1, quad(qb, k));
115                    c2 = arch::__crc32d(c2, quad(qc, k));
116                }
117            }
118            c0 = crate::combine::shift(shift, c0) ^ c1;
119            c0 = crate::combine::shift(shift, c0) ^ c2;
120        }
121    }
122
123    // Remaining words, then unaligned trailing bytes.
124    let r = crc_quads(!c0, quads);
125    crc_bytes(r, post_quad)
126}

128/// CRCs a byte slice whose length is a multiple of 8, a word at a time.
129#[target_feature(enable = "crc")]
130unsafe fn crc_quads(crc: u32, quads: &[u8]) -> u32 {
131    let mut c = !crc;
132
133    // unrolling increases performance by a lot
134    let mut quad_iter = quads.chunks_exact(64);
135    for chunk in &mut quad_iter {
136        c = arch::__crc32d(c, quad(chunk, 0));
137        c = arch::__crc32d(c, quad(chunk, 1));
138        c = arch::__crc32d(c, quad(chunk, 2));
139        c = arch::__crc32d(c, quad(chunk, 3));
140        c = arch::__crc32d(c, quad(chunk, 4));
141        c = arch::__crc32d(c, quad(chunk, 5));
142        c = arch::__crc32d(c, quad(chunk, 6));
143        c = arch::__crc32d(c, quad(chunk, 7));
144    }
145    c = quad_iter
146        .remainder()
147        .chunks_exact(8)
148        .fold(c, |acc, q| arch::__crc32d(acc, quad(q, 0)));
149
150    !c
151}

153#[target_feature(enable = "crc")]
154unsafe fn crc_bytes(crc: u32, bytes: &[u8]) -> u32 {
155    let c = bytes.iter().fold(!crc, |acc, &b| arch::__crc32b(acc, b));
156    !c
157}

161const FOLD_128: [u64; 2] = fold_keys(128);
162const FOLD_256: [u64; 2] = fold_keys(256);
163const FOLD_512: [u64; 2] = fold_keys(512);
164const FOLD_1024: [u64; 2] = fold_keys(1024);
165const FOLD_2048: [u64; 2] = fold_keys(2048);

167/// Below this, the `crc32` chains are faster than folding.
168const MIN_PMULL_BYTES: usize = 64;
169
170/// Folds `data` with as many 128-bit streams as pay for their setup and collapse at its length:
171/// each fold of a stream waits for the last, so more streams keep more `PMULL`s in flight. The
172/// crossovers are an Apple M4's. `data` holds at least `MIN_PMULL_BYTES`.
173#[target_feature(enable = "crc", enable = "aes")]
174unsafe fn calculate_pmull(crc: u32, data: &[u8]) -> u32 {
175    debug_assert!(data.len() >= MIN_PMULL_BYTES);
176    match data.len() {
177        0..=255 => fold_streams::<4>(crc, data, FOLD_512),
178        256..=4095 => fold_streams::<8>(crc, data, FOLD_1024),
179        _ => fold_streams::<16>(crc, data, FOLD_2048),
180    }
181}

183/// `N` streams of 16 bytes each are folded forward by `16 * N` bytes (`k`) with carry-less
184/// multiplies until fewer than that remain, then halved (each fold of a half into the other a
185/// level of latency, not a fold of every stream in turn) down to one, absorbing what remains of
186/// the input on the way. What is left is 16 bytes whose CRC (from a zero state), followed by the
187/// last 0 to 15 bytes, is the CRC of the whole. `data` holds at least `16 * N` bytes, and `N` is
188/// 4, 8 or 16.
189#[target_feature(enable = "crc", enable = "aes")]
190#[inline]
191unsafe fn fold_streams<const N: usize>(crc: u32, mut data: &[u8], k: [u64; 2]) -> u32 {
192    let mut v = [arch::vdupq_n_u64(0); N];
193    for x in v.iter_mut() {
194        *x = load(&mut data);
195    }
196    // The CRC state (inverted, the internal form) goes into the first 4 bytes.
197    v[0] = arch::veorq_u64(
198        v[0],
199        arch::vsetq_lane_u64::<0>(!crc as u64, arch::vdupq_n_u64(0)),
200    );
201
202    let k = keys(k);
203    while data.len() >= 16 * N {
204        for x in v.iter_mut() {
205            *x = fold(*x, load(&mut data), k);
206        }
207    }
208
209    // Fewer than `16 * N` bytes remain. Halve `2h` streams to `h` by folding each of the first
210    // half `h` blocks forward into its partner, then let the `h` take `16 * h` more bytes if
211    // there are that many: fewer than `16 * h` remain after each level. The levels are written
212    // out: as a loop, the streams no longer stay in registers.
213    macro_rules! halve {
214        ($h:expr, $keys:expr) => {
215            if N >= 2 * $h {
216                let k = keys($keys);
217                for i in 0..$h {
218                    v[i] = fold(v[i], v[i + $h], k);
219                }
220                if data.len() >= 16 * $h {
221                    for x in v[..$h].iter_mut() {
222                        *x = fold(*x, load(&mut data), k);
223                    }
224                }
225            }
226        };
227    }
228    halve!(8, FOLD_1024);
229    halve!(4, FOLD_512);
230    halve!(2, FOLD_256);
231    halve!(1, FOLD_128);
232
233    let c = arch::__crc32d(0, arch::vgetq_lane_u64::<0>(v[0]));
234    let c = arch::__crc32d(c, arch::vgetq_lane_u64::<1>(v[0]));
235    !crc_tail(c, data)
236}

238/// Advances the internal (inverted) state `c` over the last 0 to 15 bytes: at most one each of
239/// `crc32x`, `crc32w`, `crc32h` and `crc32b`, rather than a `crc32b` for each.
240#[target_feature(enable = "crc")]
241#[inline]
242unsafe fn crc_tail(mut c: u32, mut data: &[u8]) -> u32 {
243    // Often there's none: without this, the 4-stream kernel runs 64 bytes 28% slower.
244    if data.is_empty() {
245        return c;
246    }
247    if data.len() >= 8 {
248        c = arch::__crc32d(c, quad(data, 0));
249        data = &data[8..];
250    }
251    if data.len() >= 4 {
252        c = arch::__crc32w(c, u32::from_le_bytes(data[..4].try_into().unwrap()));
253        data = &data[4..];
254    }
255    if data.len() >= 2 {
256        c = arch::__crc32h(c, u16::from_le_bytes(data[..2].try_into().unwrap()));
257        data = &data[2..];
258    }
259    if let Some(&b) = data.first() {
260        c = arch::__crc32b(c, b);
261    }
262    c
263}

265/// The two fold constants as a vector, low lane first.
266#[inline(always)]
267unsafe fn keys(k: [u64; 2]) -> arch::uint64x2_t {
268    arch::vld1q_u64(k.as_ptr())
269}

271/// Folds `a` forward by the distance `k` is for and adds it to `b`: `a.lo * k.lo + a.hi * k.hi +
272/// b`, carry-less.
273#[target_feature(enable = "aes")]
274#[inline]
275unsafe fn fold(a: arch::uint64x2_t, b: arch::uint64x2_t, k: arch::uint64x2_t) -> arch::uint64x2_t {
276    let lo = arch::vmull_p64(arch::vgetq_lane_u64::<0>(a), arch::vgetq_lane_u64::<0>(k));
277    let hi = arch::vmull_high_p64(
278        arch::vreinterpretq_p64_u64(a),
279        arch::vreinterpretq_p64_u64(k),
280    );
281    arch::veorq_u64(
282        b,
283        arch::veorq_u64(
284            arch::vreinterpretq_u64_p128(lo),
285            arch::vreinterpretq_u64_p128(hi),
286        ),
287    )
288}

290/// Loads the next 16 bytes of `a`, little-endian lanes, and advances `a` past them.
291#[inline(always)]
292unsafe fn load(a: &mut &[u8]) -> arch::uint64x2_t {
293    // Indexing `a[..16]` first makes a miscounted length panic instead of reading out of bounds;
294    // the callers' loop conditions imply it, so the check compiles away.
295    let r = arch::vreinterpretq_u64_u8(arch::vld1q_u8(a[..16].as_ptr()));
296    *a = &a[16..];
297    r
298}

342                    aarch64 = kernel(aarch64, chunk);
