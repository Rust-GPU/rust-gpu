22pub(crate) const FILENAME: &str = "lib.rmeta-link";
23pub(crate) const SECTION: &str = ".rmeta-link";
24
25pub struct RmetaLink {
26    pub rust_object_files: Vec<String>,
27    /// Positionally aligned with `native_libraries` in regular metadata: index `i` is the
28    /// bundled filename for native library `i`, or `None` if that library needs no bundling.
29    pub native_lib_filenames: Vec<Option<String>>,
30}
31
32impl RmetaLink {
33    pub(crate) fn encode(&self) -> Vec<u8> {
34        let mut encoder = MemEncoder::new();
35        self.rust_object_files.encode(&mut encoder);
36        self.native_lib_filenames.encode(&mut encoder);
37        let mut data = encoder.finish();
38        data.extend_from_slice(MAGIC_END_BYTES);
39        data
40    }
41
42    pub(crate) fn decode(data: &[u8]) -> Option<RmetaLink> {
43        let mut decoder = MemDecoder::new(data, 0).ok()?;
44        let rust_object_files = Vec::<String>::decode(&mut decoder);
45        let native_lib_filenames = Vec::<Option<String>>::decode(&mut decoder);
46        Some(RmetaLink { rust_object_files, native_lib_filenames })
47    }
48}
49
50/// Reads the link-time metadata from an already-parsed archive.
51pub fn read(archive: &ArchiveFile<'_>, archive_data: &[u8], rlib_path: &Path) -> Option<RmetaLink> {
52    for entry in archive.members() {
53        let entry = entry.ok()?;
54        if entry.name() == FILENAME.as_bytes() {
55            let data = entry.data(archive_data).ok()?;
56            let section_data = search_for_section(rlib_path, data, SECTION).ok()?;
57            return RmetaLink::decode(section_data);
58        }
59    }
60    None
61}
62
63/// Like [`read`], but parses the archive from raw bytes.
64///
65/// Use this when the caller's `ArchiveFile` comes from a different version of the `object` crate.
66pub fn read_from_data(archive_data: &[u8], rlib_path: &Path) -> Option<RmetaLink> {
67    let archive = ArchiveFile::parse(archive_data).ok()?;
68    read(&archive, archive_data, rlib_path)
69}

71#[derive(Default)]
72pub struct RmetaLinkCache {
73    cache: FxHashMap<PathBuf, Option<RmetaLink>>,
74}
75
76impl RmetaLinkCache {
77    pub fn get_or_insert_with(
78        &mut self,
79        rlib_path: &Path,
80        load: impl FnOnce() -> Option<RmetaLink>,
81    ) -> Option<&RmetaLink> {
82        self.cache.entry(rlib_path.to_path_buf()).or_insert_with(load).as_ref()
83    }
84
85    pub fn native_lib_filenames(
86        &mut self,
87        target: &Target,
88        rlib_path: &Path,
89        native_libs: &[NativeLib],
90    ) -> Vec<Option<Symbol>> {
91        if !crate_may_have_bundled_libs(native_libs) {
92            return Vec::new();
93        }
94        self.get_or_insert_with(rlib_path, || read_from_path(target, rlib_path))
95            .map(|rl| {
96                rl.native_lib_filenames.iter().map(|f| f.as_deref().map(Symbol::intern)).collect()
97            })
98            .unwrap_or_default()
99    }
100}
101
102fn crate_may_have_bundled_libs(libs: &[NativeLib]) -> bool {
103    libs.iter()
104        .any(|lib| matches!(lib.kind, NativeLibKind::Static { bundle: Some(true) | None, .. }))
105}

107// FIXME: this is mostly a copy-paste of `DefaultMetadataLoader::get_rlib_metadata`.
108fn read_from_path(target: &Target, path: &Path) -> Option<RmetaLink> {
109    let Ok(file) = File::open(path) else {
110        debug!("failed to open rlib for rmeta-link: {}", path.display());
111        return None;
112    };
113    let Ok(mmap) = (unsafe { Mmap::map(file) }) else {
114        debug!("failed to mmap rlib for rmeta-link: {}", path.display());
115        return None;
116    };
117
118    if target.is_like_aix {
119        let archive = ArchiveFile::parse(&*mmap).ok()?;
120        for entry in archive.members() {
121            let entry = entry.ok()?;
122            if entry.name() == FILENAME.as_bytes() {
123                let member_data = entry.data(&*mmap).ok()?;
124                let section_data = get_metadata_xcoff(path, member_data).ok()?;
125                return RmetaLink::decode(section_data);
126            }
127        }
128        return None;
129    }
130
131    read_from_data(&mmap, path)
132}
