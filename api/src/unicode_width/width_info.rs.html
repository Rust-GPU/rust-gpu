11#[derive(Clone, Copy, Debug, PartialEq, Eq)]
12pub(crate) struct WidthInfo(u16);
13
14#[path = "gen/width_info.rs"]
15mod gen;
16
17const LIGATURE_TRANSPARENT_MASK: u16 = 0b0010_0000_0000_0000;
18
19impl WidthInfo {
20    // Additional constants are in the `gen` module.
21
22    /// Whether this width mode is ligature_transparent
23    /// (has 5th MSB set.)
24    pub(crate) fn is_ligature_transparent(self) -> bool {
25        (self.0 & 0b0000_1000_0000_0000) == 0b0000_1000_0000_0000
26    }
27
28    /// Sets 6th MSB.
29    pub(crate) fn set_zwj_bit(self) -> Self {
30        Self(self.0 | 0b0000_0100_0000_0000)
31    }
32
33    /// Has top bit set
34    pub(crate) fn is_emoji_presentation(self) -> bool {
35        (self.0 & WidthInfo::VARIATION_SELECTOR_16.0) == WidthInfo::VARIATION_SELECTOR_16.0
36    }
37
38    pub(crate) fn is_zwj_emoji_presentation(self) -> bool {
39        (self.0 & 0b1011_0000_0000_0000) == 0b1001_0000_0000_0000
40    }
41
42    /// Set top bit
43    pub(crate) fn set_emoji_presentation(self) -> Self {
44        if (self.0 & LIGATURE_TRANSPARENT_MASK) == LIGATURE_TRANSPARENT_MASK
45            || (self.0 & 0b1001_0000_0000_0000) == 0b0001_0000_0000_0000
46        {
47            Self(
48                self.0
49                    | WidthInfo::VARIATION_SELECTOR_16.0
50                        & !WidthInfo::VARIATION_SELECTOR_15.0
51                        & !WidthInfo::VARIATION_SELECTOR_1_2_OR_3.0,
52            )
53        } else {
54            Self::VARIATION_SELECTOR_16
55        }
56    }
57
58    /// Clear top bit
59    pub(crate) fn unset_emoji_presentation(self) -> Self {
60        if (self.0 & LIGATURE_TRANSPARENT_MASK) == LIGATURE_TRANSPARENT_MASK {
61            Self(self.0 & !WidthInfo::VARIATION_SELECTOR_16.0)
62        } else {
63            Self::DEFAULT
64        }
65    }
66
67    /// Has 2nd bit set
68    pub(crate) fn is_text_presentation(self) -> bool {
69        (self.0 & WidthInfo::VARIATION_SELECTOR_15.0) == WidthInfo::VARIATION_SELECTOR_15.0
70    }
71
72    /// Set 2nd bit
73    pub(crate) fn set_text_presentation(self) -> Self {
74        if (self.0 & LIGATURE_TRANSPARENT_MASK) == LIGATURE_TRANSPARENT_MASK {
75            Self(
76                self.0
77                    | WidthInfo::VARIATION_SELECTOR_15.0
78                        & !WidthInfo::VARIATION_SELECTOR_16.0
79                        & !WidthInfo::VARIATION_SELECTOR_1_2_OR_3.0,
80            )
81        } else {
82            Self(WidthInfo::VARIATION_SELECTOR_15.0)
83        }
84    }
85
86    /// Clear 2nd bit
87    pub(crate) fn unset_text_presentation(self) -> Self {
88        Self(self.0 & !WidthInfo::VARIATION_SELECTOR_15.0)
89    }
90
91    /// Has 7th bit set
92    pub(crate) fn is_vs1_2_3(self) -> bool {
93        (self.0 & WidthInfo::VARIATION_SELECTOR_1_2_OR_3.0)
94            == WidthInfo::VARIATION_SELECTOR_1_2_OR_3.0
95    }
96
97    /// Set 7th bit
98    pub(crate) fn set_vs1_2_3(self) -> Self {
99        if (self.0 & LIGATURE_TRANSPARENT_MASK) == LIGATURE_TRANSPARENT_MASK {
100            Self(
101                self.0
102                    | WidthInfo::VARIATION_SELECTOR_1_2_OR_3.0
103                        & !WidthInfo::VARIATION_SELECTOR_15.0
104                        & !WidthInfo::VARIATION_SELECTOR_16.0,
105            )
106        } else {
107            Self(WidthInfo::VARIATION_SELECTOR_1_2_OR_3.0)
108        }
109    }
110
111    /// Clear 7th bit
112    pub(crate) fn unset_vs1_2_3(self) -> Self {
113        Self(self.0 & !WidthInfo::VARIATION_SELECTOR_1_2_OR_3.0)
114    }
115}
